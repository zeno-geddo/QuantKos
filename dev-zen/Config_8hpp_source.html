QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

172 struct Heston {
173 Real k = 2.0;
174 Real theta = 0.04;
175 Real sigma = 0.3;
176 Real rho = -0.7;
177 } heston;
178
185 struct Bates : public Heston {
186 Real lambda_J = 0.1;
187 Real mu_J = -0.1;
189 Real sigma_J = 0.15;
191 } bates;
192
197 void validate() const {
198 if (id_model == KI::MathModel::Heston) {
199 validate_heston_params(heston);
200 } else if (id_model == KI::MathModel::Bates) {
201 validate_bates_params(bates);
202 }
203 }
204
212 void print(const MarketConfig &market, std::string_view indent = "") const {
213 // Assemble the equations
214 std::string str_math_model = "";
215 std::string ind(indent);
216
217 if (id_model == KI::MathModel::Heston) {
218 str_math_model = ind + "\t\t\t dSt = (r - q)·St·dt + √(vt)·St·dW1,t\n" +
219 ind + "\t\t\t dvt = κ·(θ - vt)·dt + σ·√(vt)·dW2,t\n" +
220 ind + "\t\t\t where : E[dW1,t·dW2,t] = ρ·dt\n";
221 } else if (id_model == KI::MathModel::Bates) {
222 // dNt = Merton jump process
223 str_math_model = ind + "\t\t dSt = (r - q - λ·κ_J)·St·dt + √(vt)·St·dW1,t + St⁻·(J - 1)·dNt\n" +
224 ind + "\t\t dvt = κ·(θ - vt)·dt + σ·√(vt)·dW2,t\n" +
225 ind + "\t\t where :\n" + // λ·κJ = Martingale compensator
226 ind + "\t\t ln(J) ~ N(μ_J, σ_J²),\n" +
227 // log normal jump size (J = Realized Random jump size)
228 ind + "\t\t κ_J = exp(μ_J + 0.5·σ_J²) - 1,\n" +
229 // expected percentage change in the asset price caused by a single jump
230 ind + "\t\t E[dW1,t·dW2,t] = ρ·dt\n";
231 }
232
233 if (id_model == KI::MathModel::Heston || id_model == KI::MathModel::Bates) {
234 // Handle parameter reference mapping cleanly based on active execution mode (upcast if needed)
235 const Heston &base_params = (id_model == KI::MathModel::Bates)
236 ? static_cast<const Heston &>(bates)
237 : heston;
238
239 std::cout << indent << " [" << KK::Model << "]\n"
240 << indent << " ID Model : " << id_model << "\n"
241 << indent << " --------------------------------------------------------\n"
242 << str_math_model
243 << indent << " --------------------------------------------------------\n"
244 << indent << " Core Parameters :\n"
245 << indent << " Risk-free interest rate (" << KK::MarketParams::r <<
246 ") : " << market.r << "\n"
247 << indent << " Continuous dividend yield (" << KK::MarketParams::q <<
248 ") : " << market.q << "\n"
249 << indent << " Mean reversion speed of the variance (" << KK::MathModelParams::k <<
250 ") : " << base_params.k << "\n"
251 << indent << " Mean reversion level of the variance (" << KK::MathModelParams::theta <<
252 ") : " << base_params.theta << "\n"
253 << indent << " Volatility of the variance (" << KK::MathModelParams::sigma <<
254 ") : " << base_params.sigma << "\n"
255 << indent << " Price-Variance correlation (" << KK::MathModelParams::rho <<
256 ") : " << base_params.rho << "\n";
257
258 if (2.0 * base_params.k * base_params.theta <= base_params.sigma * base_params.sigma) {
259 std::cout << indent <<
260 " Feller condition (2κθ > σ²) is violated. Variance paths may touch zero.\n";
261 } else {
262 std::cout << indent <<
263 " Feller condition (2κθ > σ²) respected.\n";
264 }
265
266 if (id_model == KI::MathModel::Bates) {
267 std::cout << indent << " Jump Parameters:\n"
268 << indent << " Jump Intensity (λ) : " << bates.lambda_J << "\n"
269 << indent << " Mean Jump Size (μJ) : " << bates.mu_J << "\n"
270 << indent << " Jump Volatility (σJ) : " << bates.sigma_J << "\n";
271 }
272 } else {
273 //const std::string msg = "[Config Error] Model selected ('" + id_model + "') is not a valid model !";
274 throw std::invalid_argument(config_err_msg(KK::Model, KK::Model,
275 "Model selected is not a valid model !"));
276 }
277 }

281 void validate_heston_params(const Heston &h) const {
282 if (h.k < 0.0)
283 throw std::invalid_argument("[Model Config] κ must be positive.");
284 if (h.theta < 0.0)
285 throw std::invalid_argument("[Model Config] θ must be positive.");
286 if (h.sigma < 0.0)
287 throw std::invalid_argument("[Model Config] σ must be positive.");
288 if (h.rho < -1.0 || h.rho > 1.0)
289 throw std::invalid_argument("[Model Config] ρ must fall within [-1.0, 1.0].");
290 }
291
293 void validate_bates_params(const Bates &b) const {
294 // Leverage inheritance! Validate the stoch-vol layer first.
295 validate_heston_params(b);
296
297 // Validate Jump Bounds
298 if (b.lambda_J < 0.0) {
299 throw std::invalid_argument("[Model Config] Bates: Jump intensity (λ) cannot be negative.");
300 }
301 if (b.sigma_J < 0.0) {
302 throw std::invalid_argument("[Model Config] Bates: Jump volatility (σJ) must be positive.");
303 }
304 }

335 struct TimeConfig {
336 Real t_end = 1.0;
337 Real inp_dt = 0.2;
338 Real dt;
339 int N_time_steps;
340
346 void validate() {
347 // Check inputs
348 if (t_end <= 0.0)
349 throw std::invalid_argument(
350 config_err_msg(KK::Time, KK::TimeParams::T_End, "must be positive."));
351 if (inp_dt <= 0.0)
352 throw std::invalid_argument(
353 config_err_msg(KK::Time, KK::TimeParams::DT, "must be positive."));
354 if (inp_dt > t_end)
355 throw std::invalid_argument(
356 config_err_msg(KK::Time, KK::TimeParams::DT, "must be smaller that total time."));
357
358 // 1. Calculate the number of steps by rounding up (ceil)
359 N_time_steps = static_cast<int>(std::ceil(t_end / inp_dt));
360
361 // 2. Uniformly distribute the time steps to fit t_end exactly
362 dt = t_end / N_time_steps;
363
364 // 3. Inform/Warn the user if their input was altered
365 // (Using a small epsilon for floating-point comparison safety)
366 if (std::abs(dt - inp_dt) > 1e-12) {
367 std::cout << "[Warning] Time step dt (" << dt
368 << "s) does not evenly divide End Time (" << t_end << ").\n"
369 << " Automatically adjusted dt to " << dt
370 << "across " << N_time_steps << " uniform steps.\n" << std::endl;
371 }
372 }
373
378 void print(const std::string_view indent = "") const {
379 std::cout << indent << " [" << KK::Time << "]\n"
380 << indent << " End Time (" << KK::TimeParams::T_End << ") : " << t_end <<
381 "\n"
382 << indent << " User Input Time Step (" << KK::TimeParams::Inp_DT << ") : " << inp_dt <<
383 "\n"
384 << indent << " Actual Time Step (" << KK::TimeParams::DT << ") : " << dt << "\n"
385 << indent << " Number Actual Time Steps (" << KK::TimeParams::N_TSteps << ") : " <<
386 N_time_steps << "\n";
387 }
388 };

396 struct MCConfig {
397 bool normalize_prices = true;
399 int N_Paths = 1000;
400 int batch_size = 0;
402 uint64_t rng_seed = 184467440737095ULL;
403 long long Max_VRAM_MB = 256;
404 long long Max_CPU_RAM_MB = 4000;
405 bool analyze_risk_neutral_payoff_distribution = false;
407 // Greeks
408 bool compute_delta_et_gamma = false;
410 bool compute_vega_et_vomma = false;
412 bool compute_vanna = false;
414 bool compute_rho = false;
416 bool compute_theta = false;
417 double spot_price_relative_bump_size = 0.001;
419 double volatility_absolute_bump_size = 0.01;
421 double risk_free_rate_absolute_bump_size = 0.0001;
423
432 double time_absolute_bump_size = 1.0 / 365.0;
433
434
441 void validate(const MarketConfig market_config) const {
442 if (N_Paths < 1)
443 throw std::invalid_argument(
444 config_err_msg(KK::MC, KK::MCParams::N_Realizations, "Must be a positive integer."));
445 if (batch_size < -1)
446 throw std::invalid_argument(config_err_msg(KK::MC, KK::MCParams::Batch_Size,
447 "Must be a positive integer if you want to impose it, -1 if you want it equal to the max number of paths, "
448 "and 0 if you want to let the machine handle this."));
449 if (batch_size > N_Paths)
450 throw std::invalid_argument(config_err_msg(
451 KK::MC, KK::MCParams::Batch_Size,
452 "The batch size must be smaller that the notal number of realizations."));
453 if (rng_seed == 0) {
454 throw std::invalid_argument(config_err_msg(KK::MC, KK::MCParams::RNG_Seed,
455 "The random number seed cannot be 0."));
456 }
457 if (rng_seed >= 18446744073709551615ULL) {
458 config_err_msg(KK::MC, KK::MCParams::RNG_Seed,
459 "[Warning] Seed is at maximum uint64 limit. Did you pass a negative number?");
460 }
461 if (Max_VRAM_MB <= 0)
462 throw std::invalid_argument(
463 config_err_msg(KK::MC, KK::MCParams::Max_VRAM_MB, "must be a positive integer."));
464 if (Max_CPU_RAM_MB <= 0)
465 throw std::invalid_argument(
466 config_err_msg(KK::MC, KK::MCParams::Max_CPU_RAM_MB, "must be a positive integer."));
467
468 if (must_compute_greeks()) {
469 validate_bump_sizes_for_greeks_computations(market_config);
470 }
471 }

475 [[nodiscard]] bool must_compute_greeks() const {
476 if (compute_delta_et_gamma || compute_vega_et_vomma || compute_rho || compute_theta) {
477 return true;
478 }
479
480 return false;
481 }

491 void validate_bump_sizes_for_greeks_computations(const MarketConfig market_config) const {
492 // 1. Spot Bump Check (Relative)
493 if (spot_price_relative_bump_size <= 0.0 || spot_price_relative_bump_size > 0.05) {
494 throw std::invalid_argument(
495 config_err_msg(KK::MC,
496 KK::MCParams::spot_price_relative_bump_size,
497 "must be strictly positive and <= 5% (0.05)."));
498 }
499
500 // 2. Volatility Bump Check (Absolute)
501 if (volatility_absolute_bump_size <= 0.0 || volatility_absolute_bump_size > 0.10) {
502 throw std::invalid_argument(
503 config_err_msg(KK::MC,
504 KK::MCParams::volatility_absolute_bump_size,
505 "must be strictly positive and <= 0.10."));
506 }
507 if (volatility_absolute_bump_size >= market_config.v0) {
508 throw std::invalid_argument(
509 config_err_msg(KK::MC,
510 KK::MCParams::volatility_absolute_bump_size,
511 "must be smaller that the initial volatility v0."));
512 }
513
514 // 3. Interest Rate Bump Check (Absolute)
515 if (risk_free_rate_absolute_bump_size <= 0.0 || risk_free_rate_absolute_bump_size > 0.05) {
516 throw std::invalid_argument(
517 config_err_msg(KK::MC,
518 KK::MCParams::volatility_absolute_bump_size,
519 "must be strictly positive and <= 0.05."));
520 }
521
522 // 4. Time Bump Check (Absolute)
523 if (time_absolute_bump_size <= 0.0 || time_absolute_bump_size > 1.0) {
524 throw std::invalid_argument(
525 config_err_msg(KK::MC,
526 KK::MCParams::time_absolute_bump_size,
527 "must be strictly positive and <= 1.0 year."));
528 }
529 }
