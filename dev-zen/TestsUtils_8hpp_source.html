64 inline std::vector<int> get_time_grid_resolutions(const int lower_exponent = 3, const int upper_exponent = 9) {
65 // Return a vector containing different resolution (Calculated as 2^i) for the grid
66 if (lower_exponent > upper_exponent)
67 throw std::runtime_error("Error : Lower exponent is greater than upper exponent");
68
69 std::vector<int> steps;
70 steps.reserve(upper_exponent - lower_exponent + 1);
71
72 for (int i = lower_exponent; i <= upper_exponent; ++i) {
73 steps.push_back(1 << i);
74 }
75
76 return steps;
77 }

110 [[nodiscard]] inline bool are_finite(const double num_price,
111 const double expected_price,
112 const double stat_error) {
113 // Reject NaNs and Infs
114 if (!std::isfinite(num_price) || !std::isfinite(expected_price) || !std::isfinite(stat_error)) {
115 return false;
116 }
117 return true;
118 }

124 [[nodiscard]] inline bool is_valid_run(const double num_price,
125 const double expected_price,
126 const double stat_error,
127 const double z_limit = 3.0) {
128 // 1. Reject NaNs and Infs immediately
129 if (!are_finite(num_price, expected_price, stat_error)) {
130 std::cout << " [ FATAL ] Non-finite numerical value detected (NaN or Inf).\n";
131 std::cout << " Num Price : " << num_price << '\n';
132 std::cout << " Reference : " << expected_price << '\n';
133 std::cout << " Stat Error : " << stat_error << '\n';
134 return false;
135 }
136
137 // 2. Reject impossible standard errors
138 if (stat_error < 0.0) {
139 std::cout << " [ FATAL ] Negative standard error detected: " << stat_error << "\n";
140 return false;
141 }
142
143 const double abs_error = std::abs(num_price - expected_price);
144
145 // 3. Deterministic Fallback (Zero Variance / No Noise)
146 if (stat_error <= 1e-14) {
147 const double strict_tol = 1e-8; // Hardcoded absolute tolerance
148 if (abs_error > strict_tol) {
149 std::cout << " [ INFO ] Deterministic test failed. Abs Error ("
150 << abs_error << ") exceeds strict limit (" << strict_tol << ").\n";
151 return false;
152 }
153 std::cout << " [ PASSED ] Deterministic results match reference exactly.\n";
154 return true;
155 }
156
157 // 4. Linear Bound Check (Avoids Division by Zero)
158 if (abs_error > z_limit * stat_error) {
159 std::cout << " [ INFO ] Price deviates by more than " << z_limit << " standard errors.\n";
160 return false;
161 }
162 std::cout << " [ PASSED ] Numerical results statistically consistent with reference.\n";
163
164 return true;
165 }

175 [[nodiscard]] inline OptionPriceErr compare_numerical_and_expected_option_price(quantkos::Config::UInputs &config,
176 const double expected_price,
177 const std::string_view indent = " ") {
178 std::cout << indent << ">>> Calling the solver ...\n";
179 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
180
181 const auto MCRes = run_simulation(config);
182 const double num_price = MCRes.OptionPrice.option_price;
183 const double stat_error = MCRes.OptionPrice.standard_error;
184 const double signed_error = num_price - expected_price;
185 const double abs_error = std::abs(signed_error);
186 const bool finite_res = are_finite(num_price, expected_price, stat_error);
187
188 std::cout << indent << "Resolution considering " << config.time.N_time_steps << " times steps:\n";
189 std::cout << indent << " Finite res : " << finite_res << '\n';
190 std::cout << indent << " Price : " << num_price << '\n';
191 std::cout << indent << " Reference : " << expected_price << '\n';
192 std::cout << indent << " Error : " << signed_error << '\n';
193 std::cout << indent << " Abs Error : " << abs_error << '\n';
194 std::cout << indent << " Std Error : " << stat_error << '\n';
195
196
197 std::string z_score_str = (stat_error > 1e-14)
198 ? std::to_string(signed_error / stat_error)
199 : "INF (Deterministic)";
200 std::cout << indent << " Z-score : " << z_score_str << '\n';
201
202 // Safe if test passed
203 constexpr double z_score_limit = 3.0;
204 const bool run_passed = is_valid_run(num_price,
205 expected_price,
206 stat_error,
207 z_score_limit);
208 if (!run_passed) {
209 std::cout << indent << "[ INFO ] Numerical results do not match reference ones. Try refining dt. \n";
210 }
211
212 // Return the results of the test and go back to tester for next text
213 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
214 std::cout << indent << ">>> Go back to the tester ...\n";
215
216 return {config.time.N_time_steps, config.time.inp_dt, signed_error, stat_error, finite_res, run_passed};
217 }

219 // Analyzes convergence data, prints diagnostics, and returns true if all Z-scores are within 3-sigma bounds.
234 [[nodiscard]] inline bool evaluate_quality_convergence_results(
235 const std::vector<OptionPriceErr> &convergence_results,
236 const std::string_view indent = " ") {
237 if (convergence_results.empty()) return false;
238
239 bool convergence_test_passed = true;
240
241 std::cout << indent << "\n::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::\n";
242 std::cout << indent << "Analyzing convergence order for given model and scheme...\n";
243 std::cout << indent <<
244 "Total Error = Discretization Error (O(dt)) + Statistical Noise (O(std/sqrt(N_paths)))\n\n";
245
246 for (size_t i = 0; i < convergence_results.size(); ++i) {
247 const auto &current = convergence_results[i];
248
249 std::string z_str = (current.stat_error > 1e-14)
250 ? std::to_string(current.error / current.stat_error)
251 : "INF";
252 std::cout << indent << "[ INFO ] Step " << i + 1
253 << ": Ndt=" << current.N_dt
254 << ", is_finite=" << current.is_finite
255 << ", ERR=" << current.error
256 << ", STAT ERR=" << current.stat_error
257 << ", Zscore=" << z_str
258 << ", within_stats=" << current.within_stat_bounds << "\n";
259
260 // Abortining it a fatal math problem is found
261 if (!current.is_finite || current.stat_error < 0.0) {
262 if (!current.is_finite) {
263 std::cerr << indent << "[ Fatal ] encountered invalid math (NaN/Inf)\n";
264 } else {
265 std::cerr << indent << "[ Fatal ] encountered invalid math (Negative StdError)\n";
266 }
267 return false;
268 }
269
270
271 // Compute convergence order only if there is a subsequent step to compare against
272 if (i < convergence_results.size() - 1) {
273 const auto &next = convergence_results[i + 1];
274 const double log_dt_diff = std::log(next.dt) - std::log(current.dt);
275 const double log_err_diff = std::log(std::abs(next.error)) - std::log(std::abs(current.error));
276 const double order = log_err_diff / log_dt_diff;
277
278 // Show convergence and warn if the error hit the statistical noise floor
279 if (std::abs(current.error) > current.stat_error) {
280 std::cout << indent << "[ INFO ] Step " << i + 1 << " -> " << i + 2
281 << " Convergence Order: " << order << "\n";
282 } else {
283 std::cout << indent << "[ INFO ] Step " << i + 1 << " -> " << i + 2
284 << " Convergence Order: " << order << " (Warning : Statistical error dominate) \n";
285 }
286 } else {
287 // Enter this block only if the last iteration (max resolution) is considered
288 // If the error exceeds 3 standard deviations from max resultion, the test mathematically fails.
289 convergence_test_passed = current.within_stat_bounds;
290 }
291 }
292
293 std::cout << indent << "::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::\n";
294
295 return convergence_test_passed;
296 }

312 [[nodiscard]] inline bool run_weak_convergence_test(
313 const std::string_view test_name,
314 KC::UInputs config,
315 const double expected_option_price,
316 const std::vector<std::pair<KI::MathModel, KI::NumScheme> > &models_to_test,
317 const std::vector<int> &time_grid_resolutions,
318 const std::string_view indent = " ") {
319 std::cout << "\n\n" << indent << "====================================================================\n"
320 << indent << " " << test_name << "\n"
321 << indent << "====================================================================\n";
322
323 std::cout << indent << "[ INFO ] Target Exact Price : " << expected_option_price << "\n";
324
325 bool all_tests_passed = true;
326
327 // Loop over models
328 for (const auto &model: models_to_test) {
329 config.model.id_model = model.first;
330 config.scheme.id_scheme = model.second;
331
332 std::vector<OptionPriceErr> convergence_results;
333
334 // Loop over time steps
335 for (int N_Tsteps: time_grid_resolutions) {
336 config.time.N_time_steps = N_Tsteps;
337 config.time.inp_dt = config.time.t_end / N_Tsteps;
338
339 std::cout << indent << "[ INFO ] MODEL : " << KI::enum_to_string(config.model.id_model) << "\n";
340 std::cout << indent << "[ INFO ] SCHEME : " << KI::enum_to_string(config.scheme.id_scheme) << "\n";
341 std::cout << indent << "[ INFO ] dt: " << std::fixed << std::setprecision(6) << config.time.inp_dt
342 << "\n";
343
344 // Run simulation
345 auto convergence_res = compare_numerical_and_expected_option_price(config, expected_option_price);
346 convergence_results.push_back(convergence_res);
347 }
348
349 // Check If numerical results are correct for this specific model/scheme combo
350 const bool model_passed = evaluate_quality_convergence_results(convergence_results);
351 if (!model_passed) {
352 all_tests_passed = false; // Mark the whole suite as failed if one combo fails
353 }
354 }
355
356 // Print final message and exit function
357 std::cout << "\n" << indent << "====================================================================\n";
358 if (all_tests_passed) {
359 std::cout << indent << "[ PASSED ] " << test_name << " successfully converged.\n";
360 } else {
361 std::cerr << indent << "[ FAILED ] " << test_name << " drifted outside statistical bounds.\n";
362 }
363 std::cout << indent << "====================================================================\n\n";
364
365 return all_tests_passed;
366 }
