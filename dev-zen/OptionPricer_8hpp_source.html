123 void accumulate_batch_payoffs(const DevBatchPayoffView &d_payoffs,
124 const int curr_batch_size) {
125 // STEP 1 : Get Batch Sum and ITM count on device
126 PayoffsBatchStats batch_stats;
127 const auto eps = eps_payoff;
128 Kokkos::parallel_reduce("Batch_Sum_And_ITM_Pass", curr_batch_size,
129 KOKKOS_LAMBDA(const int i, PayoffsBatchStats &l_stats) {
130 const double payoff = static_cast<double>(d_payoffs(i));
131 l_stats.sum += payoff; // Accumulate sum
132 if (payoff > eps) l_stats.itm += 1; // Accumulate ITM count
133 }, batch_stats);
134 const double payoff_batch_sum = batch_stats.sum;
135 const size_t batch_itm_count = batch_stats.itm;
136 const double batch_mean = payoff_batch_sum / curr_batch_size;
137
138
139 // STEP 2: Get Batch M2 (Sum of squared differences) on device---
140 double payoff_batch_sum_diff2 = 0.0;
141 Kokkos::parallel_reduce("Batch_M2_Pass", curr_batch_size,
142 KOKKOS_LAMBDA(const int i, double &sum_diff2) {
143 const double diff = static_cast<double>(d_payoffs(i)) - batch_mean;
144 sum_diff2 += (diff * diff);
145 }, payoff_batch_sum_diff2);
146
147
148 // ====================================================================
149 // --- CPU MERGE: Mean + Chan's Exact Variance ---
150 // ====================================================================
151 const size_t new_total_paths = n_payoffs_processed + curr_batch_size;
152
153 // 1. Calculate delta using the OLD global_mean (Required for payoff_sum_diff2)
154 const double delta = batch_mean - payoff_mean;
155
156 // 2. Weighted Average to update the mean
157 payoff_mean = (payoff_mean * static_cast<double>(n_payoffs_processed) +
158 batch_mean * static_cast<double>(curr_batch_size)
159 ) / static_cast<double>(new_total_paths);
160
161 // 3. Chan's Formula to update sum of diff squared (needed for variance merging)
162 // Note : M2 Chang's parallel algo : https://en.wikipedia.org/wiki/Algorithms_for_calculating_variance
163 payoff_M2_Chang += payoff_batch_sum_diff2 +
164 (delta * delta) *
165 (static_cast<double>(n_payoffs_processed * curr_batch_size)
166 / static_cast<double>(new_total_paths));
167
168 // Update trackers
169 n_payoffs_processed = new_total_paths;
170 itm_count += batch_itm_count;
171
172
173 // STEP 4 : OPTIONAL: Store arrays ONLY if distribution analysis is active ---
174 if (config.mc.analyze_risk_neutral_payoff_distribution) {
175 // 1. Destination subview (Where to put it in the global array)
176 const auto dest_slice = std::make_pair(current_path_idx, current_path_idx + curr_batch_size);
177 const auto destination_subview = Kokkos::subview(all_payoffs, dest_slice);
178
179 // 2. Source subview (Only copy the valid computed paths from this batch buffer!)
180 const auto src_slice = std::make_pair(0, curr_batch_size);
181 const auto source_subview = Kokkos::subview(d_payoffs, src_slice);
182
183 // 3. Perform device-to-device copy with matching extents
184 Kokkos::deep_copy(destination_subview, source_subview);
185 }
186 }

197 void evaluate_option_price() {
198 const auto N = static_cast<double>(config.mc.N_Paths);
199 if (N <= 0) {
200 throw std::runtime_error("Cannot compute metrics: Invalid number of paths.");
201 }
202
203 // 1. Compute discount factor (do not apply to backwards options since already applied)
204 double discount_factor = KT::real_one;
205 if (config.options.opt_type != KI::OptType::American) {
206 const auto r = static_cast<double>(config.market.r);
207 const auto t = static_cast<double>(config.time.t_end);
208 discount_factor = std::exp(-r * t); // Safely compute in double precision
209 }
210
211 // 2. Compute Unbiased sample variance: M2 / (N - 1)
212 // (We add a small ternary check to prevent division by zero if N == 1)
213 const double payoff_variance = (N > 1.0) ? (payoff_M2_Chang / (N - 1.0)) : 0.0;
214
215 // 3. Apply Discounting and scaling to get the Option Price etc.
216 // Note: standard_error : how confident we are that our calculated average is the true option price
217 const double effective_scale = discount_factor * get_scaling_factor();
218 metrics.option_price = payoff_mean * effective_scale;
219 metrics.standard_error = std::sqrt(payoff_variance / N) * effective_scale;
220 metrics.prob_itm = static_cast<double>(itm_count) / N;
221
222
223 // 4. Analyze shape of option price distribution ONLY if requested
224 if (config.mc.analyze_risk_neutral_payoff_distribution) {
225 if (all_payoffs.empty()) {
226 // Safety check just in case memory allocation failed earlier
227 throw std::runtime_error("Distribution analysis requested, but payoffs array is empty.");
228 }
229
230 const auto start_time = std::chrono::steady_clock::now();
231
232 analyze_risk_neutral_discounted_payoff_distribution(effective_scale, N);
233
234 // Store time
235 const auto end_time = std::chrono::steady_clock::now();
236 const std::chrono::duration<double> elapsed = end_time - start_time;
237 sorting_time = elapsed.count();
238 }
239
240 // Mark as successfully computed
241 metrics.is_computed = true;
242 }

316 [[nodiscard]] double get_scaling_factor() const {
317 if (config.options.opt_type == KI::OptType::BinaryCashOrNothing) {
318 // For Binary Cash, the GPU just outputs 1.0 if ITM.
319 // It is directly scaled by the requested cash payout, ignoring S0.
320 return 1.;
321 }
322
323 // For all other options, the GPU outputs normalized asset prices. We scale back by S0.
324 return config.get_price_scaling_factor();
