QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

88 // ====================================================================
89 // LEVEL 1: Dispatch the SDE Model (The Entry Point)
90 // ====================================================================
96 MCResults dispatch_model() {
97 switch (config.model.id_model) {
98 case KI::MathModel::Heston:
99 return dispatch_scheme<KI::MathModel::Heston>();
100 case KI::MathModel::Bates:
101 return dispatch_scheme<KI::MathModel::Bates>();
102 default:
103 throw std::runtime_error("Unknown SDE Model");
104 }
105 }
106
107 // ====================================================================
108 // LEVEL 2: Dispatch the Discretization Scheme
109 // ====================================================================
116 template<KI::MathModel ModelPolicy>
117 MCResults dispatch_scheme() {
118 switch (config.scheme.id_scheme) {
119 case KI::NumScheme::Euler:
120 return dispatch_opt_style<ModelPolicy, KI::NumScheme::Euler>();
121 case KI::NumScheme::ImplicitMilstein:
122 return dispatch_opt_style<ModelPolicy, KI::NumScheme::ImplicitMilstein>();
123 case KI::NumScheme::AndersonQE:
124 return dispatch_opt_style<ModelPolicy, KI::NumScheme::AndersonQE>();
125 default:
126 throw std::runtime_error("Unknown Discretization Scheme");
127 }
128 }
129
130 // ====================================================================
131 // LEVEL 3: Dispatch the Option Style (European, Asian, etc.)
132 // ====================================================================
140 template<KI::MathModel ModelPolicy, KI::NumScheme SchemePolicy>
141 MCResults dispatch_opt_style() {
142 switch (config.options.opt_type) {
143 // BASIC OPTIONS
144 case KI::OptType::European:
145 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::European>();
146 case KI::OptType::Asian:
147 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::Asian>();
148
149 // BARRIER OPTIONS
150 case KI::OptType::BarrierUpAndOut:
151 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::BarrierUpAndOut>();
152 case KI::OptType::BarrierDownAndOut:
153 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::BarrierDownAndOut>();
154 case KI::OptType::BarrierUpAndIn:
155 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::BarrierUpAndIn>();
156 case KI::OptType::BarrierDownAndIn:
157 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::BarrierDownAndIn>();
158
159 // LOOKBACK OPTIONS
160 case KI::OptType::LookbackFloatingStrike:
161 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::LookbackFloatingStrike>();
162 case KI::OptType::LookbackFixedStrike:
163 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::LookbackFixedStrike>();
164
165 // BINARY OPTIONS
166 case KI::OptType::BinaryCashOrNothing:
167 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::BinaryCashOrNothing>();
168 case KI::OptType::BinaryAssetOrNothing:
169 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::BinaryAssetOrNothing>();
170
171 // BACKWARD OPTIONS
172 case KI::OptType::American:
173 return dispatch_opt_right<ModelPolicy, SchemePolicy, KI::OptType::American>();
174
175 default:
176 throw std::runtime_error("Unknown Option Style");
177 }
178 }

181 // ====================================================================
182 // LEVEL 4: Dispatch the Option Right
183 // ====================================================================
192 template<KI::MathModel ModelPolicy, KI::NumScheme SchemePolicy, KI::OptType OptType>
193 MCResults dispatch_opt_right() {
194 switch (config.options.opt_right) {
195 case KI::OptRight::Call:
196 return execute_selected_runner<ModelPolicy, SchemePolicy, OptType, KI::OptRight::Call>();
197 case KI::OptRight::Put:
198 return execute_selected_runner<ModelPolicy, SchemePolicy, OptType, KI::OptRight::Put>();
199 default:
200 throw std::runtime_error("Unknown Option Right");
201 }
202 }
203
204 // ====================================================================
205 // LEVEL 5: Compile-Time Router
206 // All templates are now resolved. Instantiate the actual MCRunner here.
207 // ====================================================================
220 template<KI::MathModel ModelPolicy, KI::NumScheme SchemePolicy, KI::OptType OptType, KI::OptRight OptRight>
221 MCResults execute_selected_runner() {
222 if constexpr (OptType == KI::OptType::American) {
223 // If it is American, the compiler ONLY consider this branch.
224 return ForwardBackwardMCRunner<ModelPolicy, SchemePolicy, OptRight>(config).run();
225 } else {
226 // For all other options, the compiler ONLY consider this branch.
227 return ForwardMCRunner<ModelPolicy, SchemePolicy, OptType, OptRight>(config).run();
228 }
229 }
