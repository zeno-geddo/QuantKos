38 // Calculates the exponent: C + D*v0 + i*phi*ln(S0)
39 // Note : Can be reused for the bates model

56 inline Complex get_heston_characteristic_exponent(const double phi,
57 const KC::MarketConfig &m,
58 const KC::MathModelConfig::Heston &p,
59 const double T, const int j) {
60 // Promote input configuration fields to double to prevent template
61 // argument deduction failures when mixing with std::complex<double> in single-precision builds.
62 const double rho = static_cast<double>(p.rho);
63 const double sigma = static_cast<double>(p.sigma);
64 const double k = static_cast<double>(p.k);
65 const double theta = static_cast<double>(p.theta);
66
67 const double r = static_cast<double>(m.r);
68 const double q = static_cast<double>(m.q);
69 const double v0 = static_cast<double>(m.v0);
70 const double S0 = static_cast<double>(m.S0);
71
72 // Heston specific parameters for probabilities P1 and P2 (After Eq 1.40, F. Rouah)
73 const double u = (j == 1) ? 0.5 : -0.5;
74 const double b = (j == 1) ? k - rho * sigma : k;
75
76 // 1. Calculate 'd' (Eq 2.54, F. Rouah)
77 const Complex term1_d = std::pow(rho * sigma * i_unit * phi - b, 2.0);
78 const Complex term2_d = sigma * sigma * (2.0 * u * i_unit * phi - phi * phi);
79 const Complex d = std::sqrt(term1_d - term2_d);
80
81 // 2. Calculate 'c' using the Albrecher Fix (Eq 2.15, F. Rouah)
82 const Complex num_c(b - rho * sigma * i_unit * phi - d);
83 const Complex denum_c(b - rho * sigma * i_unit * phi + d);
84 const Complex c = num_c / denum_c;
85
86 // 3. Calculate 'C' (Eq 2.17, F. Rouah)
87 const Complex term1_C = (r - q) * i_unit * phi * T;
88 const Complex term2_C_fact = (k * theta) / (sigma * sigma);
89 const Complex term2_C_term1 = (b - rho * sigma * i_unit * phi - d) * T;
90 const Complex term2_C_term2 = 2.0 * std::log((1.0 - c * std::exp(-d * T)) / (1.0 - c));
91 const Complex term2_C = term2_C_fact * (term2_C_term1 - term2_C_term2);
92 const Complex C = term1_C + term2_C;
93
94 // 4. Calculate 'D' (Eq 2.14, F. Rouah)
95 const Complex D_factor1 = (b - rho * sigma * i_unit * phi - d) / (sigma * sigma);
96 const Complex D_factor2 = (1.0 - std::exp(-d * T)) / (1.0 - c * std::exp(-d * T));
97 const Complex D = D_factor1 * D_factor2;
98
99 // 5. Return eponent of the Characteristic Function f_j(phi) (Eq 1.48, F. Rouah)
100 const double x_t = std::log(S0);
101 return C + D * v0 + i_unit * phi * x_t;
102 }

104 // ========================================================================
105 // 1. THE HESTON INTEGRAND (ALBRECHER FORMULATION)
106 // ========================================================================
115 inline double heston_integrand_albrecher_formulation(double phi, const KC::UInputs &conf, int j) {
116 const auto heston_p = conf.model.heston;
117 const auto market_p = conf.market;
118 const double K = conf.options.StrikePrice;
119 const double T = conf.time.t_end;
120
121 // 1. Build the Characteristic Function f_j(phi) (Eq 1.48, F. Rouah)
122 const Complex exponent = get_heston_characteristic_exponent(phi, market_p, heston_p, T, j);
123 const Complex f_j = std::exp(exponent);
124
125 // 2. Return the real part of the final integrand (Eq 2.13, F. Rouah)
126 const Complex numerator = std::exp(-i_unit * phi * std::log(K)) * f_j;
127 const Complex denominator = i_unit * phi;
128 return std::real(numerator / denominator);
129 }

145 inline double Probability(const KC::UInputs &conf, const int j, const double phi_max = 100.0) {
146 // Create a lambda that binds the configuration and j-index, not expected in the gauss-legendre implementation
147 // The compiler should create a functor behind the scenes
148 auto integrand = [&](double phi) {
149 return heston_integrand_albrecher_formulation(phi, conf, j);
150 };
151
152 // Call the numerical integrator
153 const double integral = integrate_gl64(integrand, 0.0, phi_max);
154
155 // Compute and return the integral (Eq. , F. Rouah)
156 return 0.5 + (1.0 / M_PI) * integral;
157 }

174 inline double get_exact_eu_call_option_price(const KC::UInputs &conf, const double upper_bound = 100.) {
175 // Check that a european call is considered
176 const bool condition = (conf.options.opt_right == KI::OptRight::Call) and (
177 conf.options.opt_type == KI::OptType::European);
178 if (!condition) {
179 throw std::runtime_error(
180 "Must consider a European Call option for the Heston weak convergence test !");
181 }
182
183 const double S0 = static_cast<double>(conf.market.S0);
184 const double q = static_cast<double>(conf.market.q);
185 const double r = static_cast<double>(conf.market.r);
186 const double K = static_cast<double>(conf.options.StrikePrice);
187 const double T = static_cast<double>(conf.time.t_end);
188
189
190 // Compute the probabilities (The infinite integral is truncated at phi_max = upper_bound).
191 const double P1 = Probability(conf, 1, upper_bound);
192 const double P2 = Probability(conf, 2, upper_bound);
193
194 return S0 * std::exp(-q * T) * P1 - K * std::exp(-r * T) * P2;
195 }
