46 inline KT::Real get_exact_eu_call_option_price(const KC::UInputs &conf) {
47 // Assume considering European Call
48 const bool condition = (conf.options.opt_right == KI::OptRight::Call) and (
49 conf.options.opt_type == KI::OptType::European);
50 if (!condition) {
51 throw std::runtime_error(
52 "Must consider a European Call option for the Black Scholes weak convergence test !");
53 }
54
55 const KT::Real S = conf.market.S0;
56 const KT::Real r = conf.market.r;
57 const KT::Real q = conf.market.q;
58 const KT::Real K = conf.options.StrikePrice;
59 const KT::Real T = conf.time.t_end;
60 const KT::Real vol = std::sqrt(conf.market.v0); // Extract Volatility from Variance
61
62 const KT::Real d1 = (std::log(S / K) + (r - q + 0.5 * vol * vol) * T) / (vol * std::sqrt(T));
63 const KT::Real d2 = d1 - vol * std::sqrt(T);
64 return S * std::exp(-q * T) * normalCDF(d1) - K * std::exp(-r * T) * normalCDF(d2);
65 }
