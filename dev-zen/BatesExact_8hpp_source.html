39 // 1. THE BATES INTEGRAND (COMPOSING HESTON + MERTON JUMPS)
40 // ========================================================================
49 inline double bates_integrand_albrecher_formulation(const double phi, const KC::UInputs &conf, int j) {
50 const auto bates = conf.model.bates;
51 const auto m = conf.market;
52 const double K = conf.options.StrikePrice;
53 const double T = conf.time.t_end;
54
55 // 1. Get the continuous Heston exponent
56 // Note: 'bates' naturally upcasts to a Heston struct.
57 const Complex heston_exponent = KE::get_heston_characteristic_exponent(phi,
58 m,
59 bates,
60 T,
61 j);
62
63 // 2. Calculate the Bates Jump Components (E_j)
64 const double k_bar = std::exp(bates.mu_J + 0.5 * bates.sigma_J * bates.sigma_J) - 1.0;
65 const double u_jump = (j == 1) ? 1.0 : 0.0;
66 const double mu_J_j = (j == 1) ? bates.mu_J + bates.sigma_J * bates.sigma_J : bates.mu_J;
67
68 const Complex jump_term_inner = std::exp(i_unit * phi * mu_J_j -
69 0.5 * bates.sigma_J * bates.sigma_J * phi * phi) - 1.0;
70 const Complex E_j = bates.lambda_J * T * (-i_unit * phi * k_bar +
71 std::pow(1.0 + k_bar, u_jump) * jump_term_inner);
72
73 // 3. Compute the characteristic function (Combine heston and merton exponents)
74 const Complex f_j = std::exp(heston_exponent + E_j);
75
76 // 4. Return the real part of the final integrand
77 const Complex numerator = std::exp(-i_unit * phi * std::log(K)) * f_j;
78 const Complex denominator = i_unit * phi;
79 return std::real(numerator / denominator);
80 }

82 // ========================================================================
83 // 2. THE BATES PROBABILITY
84 // ========================================================================
96 inline double Probability(const KC::UInputs &conf, const int j, const double phi_max = 100.0) {
97 // Create a lambda that binds the configuration and j-index, not expected in the gauss-legendre implementation
98 auto integrand = [&](const double phi) {
99 return bates_integrand_albrecher_formulation(phi, conf, j);
100 };
101
102 double integral = 0.0;
103 constexpr double chunk_size = 50.0; // Keep the subdomains dense!
104 const int num_chunks = static_cast<int>(phi_max / chunk_size);
105
106 // Integrate chunk by chunk
107 for (int i = 0; i < num_chunks; ++i) {
108 const double lower = i * chunk_size;
109 const double upper = (i + 1) * chunk_size;
110 integral += integrate_gl64(integrand, lower, upper);
111 }
112
113 return 0.5 + (1.0 / M_PI) * integral;
114 }

117 // 3. BATES EU CALL OPTION PRICE
118 // ========================================================================
130 inline double get_exact_eu_call_option_price(const KC::UInputs &conf, const double upper_bound = 8000.) {
131 const bool condition = (conf.options.opt_right == KI::OptRight::Call) &&
132 (conf.options.opt_type == KI::OptType::European);
133 if (!condition) {
134 throw std::runtime_error("Must consider a European Call option for the Bates analytical test!");
135 }
136
137 const auto m = conf.market;
138 const double K = conf.options.StrikePrice;
139 const double T = conf.time.t_end;
140
141 const double P1 = Probability(conf, 1, upper_bound);
142 const double P2 = Probability(conf, 2, upper_bound);
143
144 return m.S0 * std::exp(-m.q * T) * P1 - K * std::exp(-m.r * T) * P2;
145 }
