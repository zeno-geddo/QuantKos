QuantKos v0.1.0 (Beta)

It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

172 static void evaluate(const KT::Real X, KT::Real (&L)[NBasis]) {
173 // Older form
174 //L[0] = 1.0;
175 //if constexpr (NBasis > 1) L[1] = 1.0 - X;
176 //if constexpr (NBasis > 2) L[2] = 1.0 - 2.0 * X + 0.5 * X * X;
177 //if constexpr (NBasis > 3) L[3] = 1.0 - 3.0 * X + 1.5 * X * X - (X * X * X) / 6.0;
178 //if constexpr (NBasis > 4)
179 // L[4] = 1.0 - 4.0 * X + 3.0 * X * X - (2.0 / 3.0) * X * X * X + (1.0 / 24.0) * X *
180 // X * X * X;

182 // Homer's form
183 L[0] = KT::real_one;
184
185 if constexpr (NBasis > 1) L[1] = KT::real_one - X;
186
187 if constexpr (NBasis > 2)

189

195 L[4] = ((((static_cast<KT::Real>(1.0 / 24.0) * X - static_cast<KT::Real>(2.0 / 3.0)) * X
196 + KT::real_3) * X - KT::real_4) * X)
197 + KT::real_one;
198 }

393 static void apply_final_discount(const DevView1D &d_best_future_outcomes,
394 const KT::Real discount_factor,
395 const KT::Real S0,
396 const KT::Real strike_price) {
397 const int N = d_best_future_outcomes.extent(0);
398
399 // 1. Discount the future cashflows to today and calculate their sum via parallel reduction
400 double total_discounted_cf = 0.0;
401 Kokkos::parallel_reduce("LSM_Final_Discount_And_Sum", N,
402 KOKKOS_LAMBDA(const int i, double& lsum) {
403 d_best_future_outcomes(i) *= discount_factor;
404 lsum += d_best_future_outcomes(i);
405 }, total_discounted_cf);
406
407 // 2. Calculate the statistical mean of holding the option (unconditional expectation)
408 const double mean_continuation = total_discounted_cf / static_cast<double>(N);
409
410 // 3. Calculate immediate exercise value today at S0
411 const KT::Real intrinsic_today = Payoff<OptRight>::evaluate_payoff(S0, strike_price);
412
413 // 4. If immediate exercise is optimal on average, replace all path outcomes globally
414 if (intrinsic_today > mean_continuation) {
415 Kokkos::parallel_for("LSM_Immediate_Exercise", N, KOKKOS_LAMBDA(const int i) {
416 d_best_future_outcomes(i) = intrinsic_today;
417 });
418 }
419 }

433 static RegressionSums<NBasis> compute_regression_sums_on_device(const DevView1D &d_slice_prices,
434 const DevView1D &d_best_future_outcomes,
435 const KT::Real discount_factor,
436 const KT::Real strike_price) {
437 // Compute terms M and B of linear system MC = B
438 // NOTE : Method kept public since if private cuda does not have access to it
439
440 RegressionSums<NBasis> total_sums;
441 const int N_prices = d_slice_prices.extent(0);
442
443 Kokkos::parallel_reduce("LSM_Compute_Regression_Sums", N_prices,
444 KOKKOS_LAMBDA(const int i, RegressionSums<NBasis> &regr_sum) {
445 const KT::Real S = d_slice_prices(i);
446 const KT::Real intrinsic_val = Payoff<OptRight>::evaluate_payoff(
447 S, strike_price);
448
449 if (intrinsic_val > KT::real_zero) {
450 const KT::Real Y = d_best_future_outcomes(i) * discount_factor;
451 const KT::Real X = S / strike_price;
452
453 // Evaluate the basis polynomials into a local stack buffer
454 KT::Real L[NBasis];
455 LaguerreBasis<NBasis>::evaluate(X, L);
456
457 // Update M and B using the unified templated logic
458 // The compiler should unroll these loops, making it as fast as manual coding
459 for (size_t row = 0; row < NBasis; ++row) {
460 regr_sum.B(row) += static_cast<double>(L[row]) * Y;
461 for (size_t col = 0; col < NBasis; ++col) {
462 regr_sum.M(row, col) += static_cast<double>(L[row]) * static_cast<
463 double>(L[col]);
464 }
465 }
466 regr_sum.count += 1.0;
467 }
468 }
469 ,
470 total_sums
471 );
472 return total_sums;
473 }

484 KOKKOS_INLINE_FUNCTION
485 static double evaluate_expected_val_of_holding(const Coeffs<NBasis> &coeffs,
486 const KT::Real S,
487 const KT::Real strike_price) {
488 const KT::Real X = S / strike_price;
489 KT::Real L[NBasis];
490
491 // Use the unified basis evaluator
492 LaguerreBasis<NBasis>::evaluate(X, L);
493
494 double continuation_value = 0.0;
495
496 // Accumulate the weighted sum (coeffs * basis)
497 for (size_t i = 0; i < NBasis; ++i) {
498 continuation_value += coeffs.C[i] * static_cast<double>(L[i]);
499 }
500
501 return continuation_value;
502 }

516 static Coeffs<NBasis> solve_system_on_host_cholesky(const RegressionSums<NBasis> &sums) {
517 // If not enough paths to form a basis, Use statistical mean fallback
518 if (sums.count < static_cast<double>(NBasis)) return compute_fallback_coefficients(sums);
519
520 // Tikhonov Regularization
521 // Add a tiny amount of noise (1 part per million) relative to the signal present in the data (average diagonal element)
522 // KT::Real trace = 0.0;
523 // for (size_t i = 0; i < NBasis; ++i) {
524 // trace += sums.M(i, i);
525 // }
526 // const KT::Real avg_diag = trace / static_cast<KT::Real>(NBasis);
527 // constexpr KT::Real eps_factor = std::is_same_v<KT::Real, float> ? 1e-4 : 1e-6;
528 // const KT::Real reg = std::max(eps_factor * avg_diag, eps_factor);
529
530
531 // 1. Prepare local M and B
532 constexpr double reg = 1e-8;
533 double M[NBasis][NBasis];
534 double B[NBasis];
535
536 for (size_t i = 0; i < NBasis; ++i) {
537 B[i] = sums.B(i);
538 for (size_t j = 0; j < NBasis; ++j) {
539 M[i][j] = sums.M(i, j) + ((i == j) ? reg : 0.0);
540 }
541 }
542
543 // 2. Cholesky Decomposition: M = L * L^T
544 double L[NBasis][NBasis] = {0.0}; // Here L stands for lower
545
546 constexpr double singularity_limit = 1e-14;
547
548 for (size_t i = 0; i < NBasis; ++i) {
549 for (size_t j = 0; j <= i; ++j) {
550 double s = 0.0;
551 for (size_t k = 0; k < j; ++k) s += L[i][k] * L[j][k];
552
553 if (i == j) {
554 const double val = M[i][i] - s;
555 // Use statistical mean fallback if matrix is singular/not positive-definite
556 if (val < singularity_limit) return compute_fallback_coefficients(sums);
557 L[i][i] = std::sqrt(val);
558 } else {
559 L[i][j] = (M[i][j] - s) / L[j][j];
560 }
561 }
562 }
563
564 // 3. Forward substitution: L * y = B
565 double y[NBasis];
566 for (size_t i = 0; i < NBasis; ++i) {
567 double s = 0.0;
568 for (size_t k = 0; k < i; ++k) s += L[i][k] * y[k];
569 y[i] = (B[i] - s) / L[i][i];
570 }
571
572 // 4. Backward substitution: L^T * C = y
573 Coeffs<NBasis> C;
574 for (int i = static_cast<int>(NBasis) - 1; i >= 0; --i) {
575 double s = 0.0;
576 for (size_t k = i + 1; k < NBasis; ++k) s += L[k][i] * C[k];
577 C[i] = (y[i] - s) / L[i][i];
578 }
579
580 return C;
581 }

605 static Coeffs<NBasis> compute_fallback_coefficients(const RegressionSums<NBasis> &sums) {
606 Coeffs<NBasis> fallback{};
607 if (sums.count > 0.0) {
608 // sums.B(0) is the sum of all discounted Y values.
609 // Dividing by the ITM path count gives the simple average continuation value.
610 fallback.C[0] = sums.B(0) / sums.count;
611 }
612 return fallback;
613 }
