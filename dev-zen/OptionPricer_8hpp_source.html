QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

105 void accumulate_batch_payoffs(const HostBatchPayoffView &h_payoffs, const int curr_batch_size) {
106 for (int i = 0; i < curr_batch_size; ++i) {
107 const KT::Real curr_payoff = h_payoffs(i); // Get payoff value from device
108 //all_payoffs.push_back(curr_payoff); // Store payoff to host std::vector
109 all_payoffs(current_path_idx++) = curr_payoff; // Store payoff to host kokkos view
110 const auto payoff = static_cast<double>(curr_payoff); // Cast to double befor multipying
111 total_payoff_sum += payoff;
112 total_squared_payoff_sum += (payoff * payoff); // Safely multiply payoff in double precision
113 }

125 void evaluate_option_price() {
126 const auto N = static_cast<double>(config.mc.N_Paths);
127 if (all_payoffs.empty() || N <= 0) {
128 throw std::runtime_error("Cannot compute metrics: No paths accumulated.");
129 }
130
131 const auto start_time = std::chrono::steady_clock::now();
132
133 // Compute discount factor (do not apply to backwards options since already applied)
134 double discount_factor = KT::real_one;
135 if (config.options.opt_type != KI::OptType::American) {
136 const auto r = static_cast<double>(config.market.r);
137 const auto t = static_cast<double>(config.time.t_end);
138 discount_factor = std::exp(-r * t); // Safely compute in double precision
139 }
140
141 // Compute stats
142 const double sample_mean = total_payoff_sum / N;
143
144 size_t itm_count = 0;
145 double sum_sq_diff = 0.0;
146 for (size_t i = 0; i < static_cast<size_t>(N); ++i) {
147 const KT::Real curr_payoff = all_payoffs(i);
148 const double diff = static_cast<double>(curr_payoff) - sample_mean;
149 sum_sq_diff += diff * diff;
150 if (curr_payoff > eps_payoff) itm_count++; // Count paths that survived/won
151 }
152
153 const double safe_variance = sum_sq_diff / (N - 1.0); // Unbiased sample variance
154 metrics.prob_itm = static_cast<double>(itm_count) / N;
155
156 // Apply Discounting ans scaling to get the Option Price etc.
157 const double effective_scale = discount_factor * get_scaling_factor();
158 metrics.option_price = sample_mean * effective_scale;
159 metrics.standard_error = std::sqrt(safe_variance / N) * effective_scale;
160
161 // Analyze shape of option price distribution
162 analyze_option_price_distribution(effective_scale, N);
163
164 // Mark as successfully computed
165 metrics.is_computed = true;
166
167 // Store time
168 const auto end_time = std::chrono::steady_clock::now();
169 const std::chrono::duration<double> elapsed = end_time - start_time;
170 computation_time = elapsed.count();
171 }

240 [[nodiscard]] double get_scaling_factor() {
241 if (config.options.opt_type == KI::OptType::BinaryCashOrNothing) {
242 // For Binary Cash, the GPU just outputs 1.0 if ITM.
243 // It is directly scaled by the requested cash payout, ignoring S0.
244 return 1.;
245 } else {
246 // For all other options, the GPU outputs normalized asset prices. We scale back by S0.
247 return config.get_price_scaling_factor();
248 }

257 void analyze_option_price_distribution(const double effective_scale, const double N_paths) {
258 // Sort the payoffs
259 //std::sort(std::execution::par, all_payoffs.begin(), all_payoffs.end());
260 Kokkos::sort(all_payoffs);
261
262 // Compute the percentiles
263 metrics.p0 = static_cast<double>(all_payoffs(0)) * effective_scale;
264 metrics.p1 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.01))) * effective_scale;
265 metrics.p5 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.05))) * effective_scale;
266 metrics.p10 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.10))) * effective_scale;
267 metrics.p20 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.20))) * effective_scale;
268 metrics.p30 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.30))) * effective_scale;
269 metrics.p40 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.40))) * effective_scale;
270 metrics.median = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.50))) * effective_scale;
271 metrics.p60 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.60))) * effective_scale;
272 metrics.p70 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.70))) * effective_scale;
273 metrics.p80 = static_cast<double>(all_payoffs(static_cast<size_t>(N_paths * 0.80))) * effective_scale;
