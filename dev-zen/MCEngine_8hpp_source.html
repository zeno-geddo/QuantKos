It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

215 void print_progress_bar(const int n_current_batch) {
216 constexpr int bar_width = 20;
217 const int n_total_batches = batch_mem.total_batch_loops();
218
219 const float progress = static_cast<float>(n_current_batch) / static_cast<float>(n_total_batches);
220 const int bar_front = static_cast<int>(static_cast<float>(bar_width) * progress);
221 const int percentage_done = static_cast<int>(progress * 100.0f);
222
223 std::cout << " [";
224 for (int i = 0; i < bar_width; ++i) {
225 if (i < bar_front) std::cout << "#";
226 else if (i == bar_front) std::cout << ">";
227 else std::cout << ".";
228 }
229
230 std::cout << "] " << percentage_done << "% "
231 << "(" << n_current_batch << "/" << n_total_batches << " Batches)";
232
233 if (n_current_batch > 0) {
234 const auto current_time = std::chrono::steady_clock::now();
235 const std::chrono::duration<double> elapsed = current_time - start_time;
236 const double elapsed_seconds = elapsed.count();
237
238 const double total_estimated_time = elapsed_seconds / progress;
239 const double eta_seconds = total_estimated_time - elapsed_seconds;
240
241 std::cout << std::fixed << std::setprecision(1)
242 << "(T: " << elapsed_seconds << "s | "
243 << "ETA: " << (n_current_batch == n_total_batches ? 0.0 : eta_seconds) << "s)\r";
244 }
245 std::cout << "\r" << std::flush;
246 }

300 void update_progress() {
301 steps_completed++;
302
303 // Calculate current integer percentage
304 if (total_lsm_steps <= 0) return; // Prevent division by zero if there are almost no time steps
305 const int current_percent = static_cast<int>(
306 (static_cast<float>(steps_completed) / static_cast<float>(total_lsm_steps)) * 100.0f
307 );
308
309 // Print the progress bar if target percentage reached
310 if (current_percent >= last_printed_percent + update_interval_percent) {
311 print_progress_bar(steps_completed);
312 last_printed_percent = current_percent; // Update the threshold
313 }
314 }
315
319 void finalize_tracking() {
320 print_progress_bar(total_lsm_steps);
321 std::cout << "\n\n ========================================================\n" << std::endl;
322 }

423 template<typename SolverType, typename HostDeepCopy>
424 inline void run_forward_all_mc_batches(
425 PathsMCBatchMem &BatchMem,
426 const RNGManager &RNGen,
427 const SolverType &Solver,
428 KIO::OutputManager &OWriter,
429 ForwardMCProgressTracker &MCTracker,
430 HostDeepCopy &&custom_host_deep_copy // Lambda function to copy desired data to host
431 ) {
432 const int full_batch_size = BatchMem.n_sims_per_batch;
433 const int n_full_batches = BatchMem.n_full_batches();
434 const int n_sims_left_over = BatchMem.n_sims_left_over_after_full_batches();
435 const int n_total_batch_loops = BatchMem.total_batch_loops();
436
437 MCTracker.start_tracking();
438 for (int n_curr_batch = 0; n_curr_batch < n_total_batch_loops; ++n_curr_batch) {
439 const int current_batch_size = (n_curr_batch < n_full_batches) ? full_batch_size : n_sims_left_over;
440
441 // 1. Fire parallel integration kernels on the active Kokkos execution space (Device/GPU)
442 Solver.execute_batch(current_batch_size, BatchMem, RNGen);
443
444 // 2. Execute the custom host-device synchronization policy (e.g. DMA transfers, accumulation)
445 custom_host_deep_copy(n_curr_batch, current_batch_size); // Note : Injected via Lambda
446
447 // 3. Write path diagnostics/trajectories to the file system if configured
448 OWriter.save_paths_batch_if_needed(current_batch_size, BatchMem); //Save batch to disk
449
450 // 4. Advance progress and compute linear clock ETA using the tracker defined in the Canvas
451 MCTracker.update_progress(n_curr_batch + 1);
452 }
453 MCTracker.finalize_tracking();
454 }

494 template<typename MemoryType, typename PricingCallable>
495 inline MCResults execute_mc_framework(const KC::UInputs &config, PricingCallable &&pricing_function) {
496 // 1. Allocate Heavy Memory ONCE for the exact type requested
497 MemoryType Mem(config);
498
499 // 2. Get the option Price
500 OptionPricer::MCOpPrices ref_option_price_data = pricing_function(config, Mem);
501
502
503 // 3. Compute Greeks if needed (Using the generic pricing callable)
504 MCGreeks greeks{};
505 if (config.must_compute_greeks()) {
506 KC::UInputs base_greek_cfg = config; // Create a copy where spoofing params common to all greeks
507 base_greek_cfg.mc.analyze_risk_neutral_payoff_distribution = false; // payoffs never stored and sorted
508 base_greek_cfg.output.filename_paths_out = ""; // So that paths for the greeks are never saved
509 // should introduce and rewrite a flag that tell if to print or not, and during the greeks should be low verbosity
510
511 if (base_greek_cfg.mc.compute_delta_et_gamma) {
512 std::cout << "\n\n >>> Computing Delta & Gamma...\n";
513 // Get initial price and price bump considering that it could have been normalized
514 const auto S0 = base_greek_cfg.market.S0;
515 const auto h_S = S0 * base_greek_cfg.mc.spot_price_relative_bump_size;
516 auto h_S_real_scale = h_S;
517 if (base_greek_cfg.mc.normalize_prices) {
518 const auto S0_real_scale = base_greek_cfg.get_price_scaling_factor();
519 h_S_real_scale = S0_real_scale * base_greek_cfg.mc.spot_price_relative_bump_size;
520 }
521
522 KC::UInputs cfg_up = base_greek_cfg;
523 cfg_up.market.S0 = S0 + h_S;
524 double p_up = pricing_function(cfg_up, Mem).option_price;
525
526 KC::UInputs cfg_down = base_greek_cfg;
527 cfg_down.market.S0 = S0 - h_S;
528 double p_down = pricing_function(cfg_down, Mem).option_price;
529
530 // NOTE : Divide by the bump in the real scale to match the scale of the real output prices
531 greeks.delta = (p_up - p_down) / (2.0 * h_S_real_scale);
532 greeks.gamma = (p_up - 2.0 * ref_option_price_data.option_price + p_down) /
533 (h_S_real_scale * h_S_real_scale);
534 }
535
536 if (base_greek_cfg.mc.compute_vega_et_vomma) {
537 // Approach 1
538 std::cout << "\n\n >>> Computing Vega...\n";
539 const double var0 = base_greek_cfg.market.v0;
540 const double vol0 = std::sqrt(var0);
541 const double h_vol = base_greek_cfg.mc.volatility_absolute_bump_size;
542
543 KC::UInputs cfg_up = base_greek_cfg;
544 const double vol_up = vol0 + h_vol;
545 const double var_up = vol_up * vol_up;
546 cfg_up.market.v0 = var_up; // variance instead of volatility required
547 double p_up = pricing_function(cfg_up, Mem).option_price;
548
549 KC::UInputs cfg_down = base_greek_cfg;
550 const double vol_down = vol0 - h_vol;
551 const double var_down = vol_down * vol_down;
552 cfg_down.market.v0 = var_down; // variance instead of volatility required
553 double p_down = pricing_function(cfg_down, Mem).option_price;
554
555 greeks.vega = (p_up - p_down) / (2.0 * h_vol);
556 greeks.vomma = (p_up - 2.0 * ref_option_price_data.option_price + p_down) / (h_vol * h_vol);
557 }
558
559 if (base_greek_cfg.mc.compute_rho) {
560 std::cout << "\n\n >>> Computing Rho...\n";
561 const double r = base_greek_cfg.market.r;
562 const double h_r = base_greek_cfg.mc.risk_free_rate_absolute_bump_size;
563
564 KC::UInputs cfg_up = base_greek_cfg;
565 cfg_up.market.r = r + h_r;
566 double p_up = pricing_function(cfg_up, Mem).option_price;
567
568 KC::UInputs cfg_down = base_greek_cfg;
569 cfg_down.market.r = r - h_r;
570 double p_down = pricing_function(cfg_down, Mem).option_price;
571
572 greeks.rho = (p_up - p_down) / (2.0 * h_r);
573 }
574
575 if (base_greek_cfg.mc.compute_theta) {
576 std::cout << "\n\n >>> Computing Theta...\n";
577 const double T = base_greek_cfg.time.t_end; // time to maturity
578 const double h_T = base_greek_cfg.mc.time_absolute_bump_size;
579
580 // Safety check: Ensure the option has more time left than the bump size
581 if (T > h_T) {
582 KC::UInputs cfg_down = base_greek_cfg;
583
584 // Continuous Maturity Shift: Shrink T, but keep N_time_steps the same
585 // NOTE: To price the option from the perspective of tomorrow using a simulation that starts at $0$,
586 // we set the total simulation horizon ($t_{\text{end}}$) to the remaining time: $T - h_T$.
587 // In other words, we are standing at higher unit time (e.g. tomorrow), so, because 1 unit time has elapsed,
588 // the time remaining until maturity is now T−ht
589 cfg_down.time.t_end = T - h_T;
590
591 // Manually compress the continuous grid step to fit the new time, but keep N_time_steps the same
592 cfg_down.time.dt = cfg_down.time.t_end / base_greek_cfg.time.N_time_steps;
593
594 // Now run the pricing function with the compressed grid (the grid has the same shape)
595 double p_down = pricing_function(cfg_down, Mem).option_price;
596
597 // Annualized Theta: (Price with perspective of tomorrow - Price with perspective of today) / time elapsed
598 greeks.theta = (p_down - ref_option_price_data.option_price) / h_T;
599 } else {
600 std::cout <<
601 " [Warning] Option too close to expiry to compute finite-difference Theta. Theta set to zero.\n";
602 greeks.theta = 0.0;
603 }
604 }
605
606 greeks.print_summary(); // print the summary of the greeks computed
607 }
608
609 // 4. Return combined results
610 return MCResults{
611 .MCConfig = config,
612 .OptionPrice = ref_option_price_data,
613 .Greeks = greeks
614 };
615 }
