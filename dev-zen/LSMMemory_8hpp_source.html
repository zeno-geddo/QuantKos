QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

72 explicit BackwardLSMMemory(const KC::UInputs &conf) : BatchMem(conf) {
73 const int N = conf.mc.N_Paths;
74 const int T = conf.time.N_time_steps;
75
76 // 1. Calculate the RAM needed for the Master Matrix
77 const double required_master_mb = static_cast<double>(N) * T * sizeof(KT::Real) / (1024.0 * 1024.0);
78
79 // 2. Add the RAM already claimed by the Host-side batch mirrors
80 const double tot_host_batch_mb = BatchMem.bytes_to_mb(BatchMem.tot_host_memory_bytes());
81 const double host_paths_batch_mb = BatchMem.bytes_to_mb(BatchMem.host_paths_memory_bytes());
82 const double host_payoffs_batch_mb = BatchMem.bytes_to_mb(BatchMem.host_payoffs_memory_bytes());
83
84 // 3. Check if there is enough memory
85 const double total_required_cpu_mb = required_master_mb + tot_host_batch_mb;
86 const double budget_ram_mb = static_cast<double>(conf.mc.Max_CPU_RAM_MB);
87 const size_t ram_available_from_os_mb = get_available_memory_from_os_mb();
88 const double effective_memory_limit_mb = std::min(budget_ram_mb, static_cast<double>(ram_available_from_os_mb));
89 if (total_required_cpu_mb > effective_memory_limit_mb){
90 throw std::runtime_error(
91 "[Memory Capacity Error] The Longstaff-Schwartz algorithm requires CPU RAM "
92 "that, summed to the batches memory used for the forward computation, exceeds your assigned Max_CPU_RAM_MB budget or the available current RAM budget.\n"
93 " -> Master Matrix RAM : " + std::to_string(required_master_mb) + " MB\n"
94 " -> Host Paths Batch Buffers : " + std::to_string(host_paths_batch_mb) + " MB\n"
95 " -> Host Payoff BatchBuffers : " + std::to_string(host_payoffs_batch_mb) + " MB\n"
96 " -> Total Required : " + std::to_string(total_required_cpu_mb) + " MB\n"
97 " -> Config Budget : " + std::to_string(budget_ram_mb) + " MB\n"
98 " -> Available RAM (OS query) : " + std::to_string(ram_available_from_os_mb) + " MB\n"
99 "Possible Actions: \n"
100 "\t 1) Increase 'Max_CPU_RAM_MB' in your config (if more RAM is available),\n"
101 "\t 2) Reduce the the batch size (explicitly assign a moderate number of sims per batch),\n"
102 "\t 3) Reduce the number of paths/time steps.\n"
103 );
104 }
105
106 // 4. Allocate memory if there is enough
107 std::cout << " [RAM Memory] Available memory on RAM " << ram_available_from_os_mb << "MB" << std::endl;
108 std::cout << " [LSM Memory] Allocating " << N << "x" << T << " Master Matrix in Host RAM..." << std::endl;
109 h_master_paths = HostMasterPathsView("HostMasterMatrix", N, T);
110
111 std::cout << " [LSM Memory] Allocating Temporary Streaming Buffers for Regressions..." << std::endl;
112 d_prices_current_time = Kokkos::View<KT::Real *, Kokkos::LayoutLeft>("Device_Time_Slice", N);
113 d_best_future_outcomes = Kokkos::View<KT::Real *, Kokkos::LayoutLeft>("Device_Cash_Flows", N);
114 h_best_future_outcomes = Kokkos::create_mirror_view(d_best_future_outcomes);
115 }

138 void copy_current_mc_batch_to_master_mc_matrix(const int batch_idx, const int current_batch_size) {
139 // Aim : Extract the exact sub-block of the master matrix we want to fill and fill it
140 // Note: master matrix is always layout right, while batch-view depends on how the device is being used
141 // Note: subview only gives a window, it does not allocate any new memory
142
143 // 0. Move data to host using PCIe BUS, it is the slowest copy (cannot copy directly from gpu to master matrix in cpu)
144 BatchMem.deep_copy_to_host();
145
146 // 1. Create Target: CPU Master Matrix subview
147 const int path_start_idx = batch_idx * BatchMem.n_sims_per_batch;
148 const int paths_end_idx = path_start_idx + current_batch_size;
149 const auto target_paths_range = std::make_pair(path_start_idx, paths_end_idx);
150 auto h_sub_master = Kokkos::subview(h_master_paths, // target memory to slice
151 target_paths_range, // target first dimension range (the rows/paths)
152 Kokkos::ALL() // target all second dimension (all cols/times)
153 );
154
155 // 2. Create Source: CPU Host Batch subview
156 auto h_sub_batch = Kokkos::subview(BatchMem.h_batch_view,
157 std::make_pair(0, current_batch_size),
158 Kokkos::ALL()); // Handle the case of the last incomplete batch
159
160 // 3. Perform the CPU-to-CPU copy, implicitly making a transpose if necessary
161 // Note : This is much faster since it does not require PCIe BUS but is done in CPU RAM and L3 cache, they are much faster!
162 // Doing the second copy should not be a disaster in terms of performance
163 Kokkos::deep_copy(h_sub_master, h_sub_batch);
164 }

169 void bring_host_prices_time_slice_to_device(const int time_step) const {
170 // 1. Slice out the target column from the column-major master matrix.
171 // Note: Kokkos::ALL() ensures we grab every simulated path for this time step.
172 auto h_column_view = Kokkos::subview(h_master_paths, Kokkos::ALL(), time_step);
173
174 // 2. Stream the contiguous memory slice across the PCIe bus to the GPU.
175 Kokkos::deep_copy(d_prices_current_time, h_column_view);
176 }
177
179 void bring_device_cash_flows_to_host() {
180 Kokkos::deep_copy(h_best_future_outcomes, d_best_future_outcomes);
181 }
