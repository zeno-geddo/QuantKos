QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

109 inline double compute_greek_tol(
110 const double base_option_price_se,
111 const double greek_scale,
112 const double base_option_price_se_multiplier = 3.0,
113 const double relative_tolerance = 0.01,
114 const double absolute_floor = 1e-4)
115 {
116 const double scale_tolerance = relative_tolerance * std::max(std::abs(greek_scale), 1.0);
117
118 const double mc_tolerance = base_option_price_se_multiplier * base_option_price_se;
119
120 return std::max({scale_tolerance, mc_tolerance, absolute_floor});
121 }

133 inline bool check_parity(const std::string &name,
134 const double num,
135 const double expected,
136 const double tol,

144 return false;
145 }
146 std::cout << " [ PASSED ]\n";

176 inline bool run_test() {
177 constexpr std::string_view indent = " ";
178 auto config = getDefaultConfigGreekParity();
179 config.validate();
180
181 std::cout << "\n\n" << indent << ">>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>\n";
182 std::cout << indent << "[ INFO ] Initiating Greek Put-Call Parity Test \n";
183
184 // 1. Run CALL Simulation
185 config.options.opt_right = KI::OptRight::Call;
186 std::cout << indent << ">>> Running CALL configuration...\n";
187 const auto call_res = KTU::run_simulation(config);
188
189 // 2. Run PUT Simulation
190 config.options.opt_right = KI::OptRight::Put;
191 std::cout << indent << ">>> Running PUT configuration...\n";
192 const auto put_res = KTU::run_simulation(config);
193
194 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
195 std::cout << indent << ">>> Evaluating Greek's Call-Put Parities ...\n";
196
197 // Target Expected Values based on continuous-time finance
198 const double K = config.options.StrikePrice;
199 const double T = config.time.t_end;
200 const double r = config.market.r;
201 const double discount = std::exp(-r * T);
202
203 constexpr double expected_delta_diff = 1.0;
204 constexpr double expected_gamma_diff = 0.0;
205 constexpr double expected_vega_diff = 0.0;
206 constexpr double expected_vomma_diff = 0.0;
207 const double expected_rho_diff = K * T * discount;
208 const double expected_theta_diff = -r * K * discount;
209
210 // Computed Values
211 const auto &C = call_res.Greeks;
212 const auto &P = put_res.Greeks;
213
214 const double num_delta_diff = C.delta - P.delta;
215 const double num_gamma_diff = C.gamma - P.gamma;
216 const double num_vega_diff = C.vega - P.vega;
217 const double num_vomma_diff = C.vomma - P.vomma;
218 const double num_rho_diff = C.rho - P.rho;
219 const double num_theta_diff = C.theta - P.theta;
220
221 // Pre-compute practical tolerance for each Greek
222 const double base_se = call_res.OptionPrice.standard_error; // Extract base standard error from the option price
223 const double tol_delta = compute_greek_tol(base_se, 1., 3.3);
224 const double tol_gamma = compute_greek_tol(base_se, 1., 3.3);
225 const double tol_vega = compute_greek_tol(base_se, 1., 3.3);
226 const double tol_vomma = compute_greek_tol(base_se, 1., 8.);
227 const double tol_rho = compute_greek_tol(base_se, std::abs(expected_rho_diff), 3.3);
228 const double tol_theta = compute_greek_tol(base_se, std::abs(expected_theta_diff), 5.); // Higher bias for grid shift
229
230 // Check if the parities are ok
231 const bool all_tests_passed = std::min({
232 check_parity("Delta", num_delta_diff, expected_delta_diff, tol_delta, indent),
233 check_parity("Gamma", num_gamma_diff, expected_gamma_diff, tol_gamma, indent),
234 check_parity("Vega ", num_vega_diff, expected_vega_diff, tol_vega, indent),
235 check_parity("Vomma ", num_vomma_diff, expected_vomma_diff, tol_vomma, indent),
236 check_parity("Rho ", num_rho_diff, expected_rho_diff, tol_rho, indent),
237 check_parity("Theta", num_theta_diff, expected_theta_diff, tol_theta, indent)
238 }
239 );
240
241 if (all_tests_passed) {
242 std::cout << "\n" << indent << "[ SUCCESS ] All Greek finite-difference engines are structurally sound.\n";
243 } else {
244 std::cerr << "\n" << indent << "[ FAILURE ] One or more Greeks violated Put-Call Parity bounds!\n";
245 }
246
247 std::cout << indent << "<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<\n";
248
249 return all_tests_passed;
250 }
