QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

53 explicit RNGManager(const KC::UInputs &config) {
54 // Kokkos allocates a massive, flat array of independent mathematical state registers directly inside the device memory space (VRAM or system RAM).
55 // The number of states allocated is chosen by Kokkos to match the maximum physical hardware concurrency of your chip (often tens of thousands of individual slots).
56 // Each slot is initialized with a slightly scrambled version of your seed.
57 const uint64_t initial_seed = config.mc.rng_seed;
58 global_rng_pool = GlobalRNGPool(initial_seed);
59 std::cout << " [RNGenerator] Global Random Number Pool initialized correctly with seed : "

202 template<typename LocalRNGType, bool skip_second = false>
203 struct NormalPair {
206 KOKKOS_INLINE_FUNCTION RVPair operator()(LocalRNGType &local_rg) const {
207 auto pair = RVPair{.Z1 = KT::real_zero, .Z2 = KT::real_zero};
208
209 KOKKOS_IF_ON_DEVICE((
210 // GPU Box-Muller Code
211 const KT::Real U1 = KT::real_one - get_uniform<LocalRNGType>(local_rg);
212 const KT::Real U2 = get_uniform<LocalRNGType>(local_rg);
213 const KT::Real R = Kokkos::sqrt(-KT::real_two * Kokkos::log(U1));
214 const KT::Real theta = KT::real_2PI * U2;
215 pair.Z1 = R * Kokkos::cos(theta);
216 // Compute second rv only if needed
217 if constexpr (!skip_second) {
218 pair.Z2 = R * Kokkos::sin(theta);
219 }
220 ))
221
222 KOKKOS_IF_ON_HOST((
223 // CPU Marsaglia Code
224 KT::Real u, v, s;
225 do {
226 u = KT::real_two * get_uniform<LocalRNGType>(local_rg) - KT::real_one;
227 v = KT::real_two * get_uniform<LocalRNGType>(local_rg) - KT::real_one;
228 s = u * u + v * v;
229 } while (s >= KT::real_one || s == KT::real_zero);
230
231 const KT::Real multiplier = Kokkos::sqrt(-KT::real_two * Kokkos::log(s) / s);
232 pair.Z1 = u * multiplier;
233 // Compute second rv only if needed
234 if constexpr (!skip_second) {
235 pair.Z2 = v * multiplier;
236 }
237
238 ))
239
240 return pair;
241 }
242 };
