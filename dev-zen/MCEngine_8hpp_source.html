QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

184 void print_progress_bar(const int n_current_batch) {
185 constexpr int bar_width = 20;
186 const int n_total_batches = batch_mem.total_batch_loops();
187
188 const float progress = static_cast<float>(n_current_batch) / static_cast<float>(n_total_batches);
189 const int bar_front = static_cast<int>(static_cast<float>(bar_width) * progress);
190 const int percentage_done = static_cast<int>(progress * 100.0f);
191
192 std::cout << " [";
193 for (int i = 0; i < bar_width; ++i) {
194 if (i < bar_front) std::cout << "#";
195 else if (i == bar_front) std::cout << ">";
196 else std::cout << ".";
197 }
198
199 std::cout << "] " << percentage_done << "% "
200 << "(" << n_current_batch << "/" << n_total_batches << " Batches)";
201
202 if (n_current_batch > 0) {
203 const auto current_time = std::chrono::steady_clock::now();
204 const std::chrono::duration<double> elapsed = current_time - start_time;
205 const double elapsed_seconds = elapsed.count();
206
207 const double total_estimated_time = elapsed_seconds / progress;
208 const double eta_seconds = total_estimated_time - elapsed_seconds;
209
210 std::cout << std::fixed << std::setprecision(1)
211 << "(T: " << elapsed_seconds << "s | "
212 << "ETA: " << (n_current_batch == n_total_batches ? 0.0 : eta_seconds) << "s)\r";
213 }
214 std::cout << "\r" << std::flush;
215 }
216 };

264 void update_progress() {
265 steps_completed++;
266
267 // Calculate current integer percentage
268 if (total_lsm_steps <= 0) return; // Prevent division by zero if there are almost no time steps
269 const int current_percent = static_cast<int>(
270 (static_cast<float>(steps_completed) / static_cast<float>(total_lsm_steps)) * 100.0f
271 );
272
273 // Print the progress bar if target percentage reached
274 if (current_percent >= last_printed_percent + update_interval_percent) {
275 print_progress_bar(steps_completed);
276 last_printed_percent = current_percent; // Update the threshold
277 }
278
279 }

284 void finalize_tracking() {
285 print_progress_bar(total_lsm_steps);
286 std::cout << "\n\n ========================================================\n" << std::endl;
287 }

299
300 // Prevent division by zero if there are almost no time steps

303 const float progress = static_cast<float>(current_step) / static_cast<float>(total_lsm_steps);
304 const int bar_front = static_cast<int>(static_cast<float>(bar_width) * progress);
305 const int percentage_done = static_cast<int>(progress * 100.0f);

307 std::cout << " [";

309 if (i < bar_front) std::cout << "#";
310 else if (i == bar_front) std::cout << "<"; // Use a backward arrow for aesthetics
311 else std::cout << ".";
312 }
313
314 std::cout << "] " << percentage_done << "% "
315 << "(" << current_step << "/" << total_lsm_steps << " Time Slices)";
316
317 if (current_step > 0) {
318 const auto current_time = std::chrono::steady_clock::now();
319 const std::chrono::duration<double> elapsed = current_time - start_time;
320 const double elapsed_seconds = elapsed.count();
321
322 const double total_estimated_time = elapsed_seconds / progress;
323 const double eta_seconds = total_estimated_time - elapsed_seconds;
324
325 std::cout << std::fixed << std::setprecision(1)
326 << "(T: " << elapsed_seconds << "s | "
327 << "ETA: " << (current_step == total_lsm_steps ? 0.0 : eta_seconds) << "s)\r";

387 template<typename SolverType, typename HostDeepCopy>
388 inline void run_forward_all_mc_batches(
389 PathsMCBatchMem &BatchMem,
390 const RNGManager &RNGen,
391 const SolverType &Solver,
392 KIO::OutputManager &OWriter,
393 ForwardMCProgressTracker &MCTracker,
394 HostDeepCopy &&host_deep_copy // Lambda function to copy desired data to host
395 ) {
396 // [ HOST (CPU) ] [ DEVICE (GPU) ]
397 //
398 // MCRunner Loop Fires
399 // │
400 // ▼
401 // SDESolver::execute_batch()
402 // │
403 // ▼
404 // Instantiate IntegrationKernel
405 // (Flattens data onto CPU Stack)
406 // │
407 // ▼
408 // Kokkos::parallel_for() =======[ PCIe Bus Pass ]=======> GPU Spawns N Threads
409 // │
410 // ▼
411 // kernel.operator()(n_p)
412 // │
413 // ▼
414 // Time Loop (0 to N_Steps)
415 // │
416 // ▼
417 // scheme.evolve_step()
418 // │
419 // ▼
420 // Coalesced VRAM Write
421
422
423 const int full_batch_size = BatchMem.n_sims_per_batch;
424 const int n_full_batches = BatchMem.n_full_batches();
425 const int n_sims_left_over = BatchMem.n_sims_left_over_after_full_batches();
426 const int n_total_batch_loops = BatchMem.total_batch_loops();
427
428 MCTracker.start_tracking();
429 for (int b = 0; b < n_total_batch_loops; ++b) {
430 const int current_batch_size = (b < n_full_batches) ? full_batch_size : n_sims_left_over;
431
432 // 1. Fire parallel integration kernels on the active Kokkos execution space (Device/GPU)
433 Solver.execute_batch(current_batch_size, BatchMem, RNGen);
434
435 // 2. Execute the custom host-device synchronization policy (e.g. DMA transfers, accumulation)
436 host_deep_copy(b, current_batch_size); // Note : Injected via Lambda
437
438 // 3. Write path diagnostics/trajectories to the file system if configured
439 OWriter.save_paths_batch_if_needed(current_batch_size, BatchMem); //Save batch to disk
440
441 // 4. Advance progress and compute linear clock ETA using the tracker defined in the Canvas
442 MCTracker.update_progress(b + 1);
443 }
444 MCTracker.finalize_tracking();
445 }
