QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

326 void allocate_batch_memory() {
327 // 1. Determine the layout strategy chosen by the config/auto-tuner
328 if (config.mc.batch_size == -1) {
329 n_sims_per_batch = config.mc.N_Paths;
330 } else if (config.mc.batch_size == 0) {
331 n_sims_per_batch = determine_optimal_batch_size();
332 } else {
333 n_sims_per_batch = config.mc.batch_size;
334 }
335
336 // 2. High-Bound Safety: Optimization clamp (No error needed)
337 if (n_sims_per_batch > config.mc.N_Paths) {
338 n_sims_per_batch = config.mc.N_Paths;
339 }
340
341 // 3. Low-Bound Safety: Throw a hard exception if the budget is unrunnable
342 if (n_sims_per_batch < 1) {
343 throw std::runtime_error(
344 "[Critical Error] The assigned VRAM/RAM hardware budget is too restrictive "
345 "to accommodate even a single Monte Carlo path timeline simulation. "
346 "Please increase your memory budget or increase your time step size (dt)."
347 );
348 }
349
350 // Show available RAM
351 std::cout << " [OS Memory Query] Available Host RAM " << get_available_memory_from_os_mb() << "MB\n";
352
353 // Allocate the PATHS views using our newly stored class attributes
354 if (require_paths_allocated) {
355 std::cout << " [Batch Memory] Allocating reusable buffers ("
356 << n_sims_per_batch << " x " << config.time.N_time_steps <<
357 ") for the paths of the MC batches...\n";
358 d_batch_view = DevBatchPathsView("gpu_paths_batch_buffer", n_sims_per_batch, config.time.N_time_steps);
359 h_batch_view = Kokkos::create_mirror_view(d_batch_view);
360 }
361
362
363 // Allocate the PAYOFFS views
364 std::cout << " [Batch Memory] Allocating reusable buffers ("
365 << n_sims_per_batch << ") for the payoffs of the MC batches...\n";
366 d_payoffs = DevBatchPayoffView("gpu_payoffs_batch_buffer", n_sims_per_batch);
367 // h_payoffs = Kokkos::create_mirror_view(d_payoffs);
368 }

374 [[nodiscard]] int determine_optimal_batch_size() const {
375 const double bytes_per_sde_path = config.time.N_time_steps * sizeof(KT::Real);
376
377#if defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP) || defined(KOKKOS_ENABLE_SYCL)
378 const double gpu_budget_bytes = static_cast<double>(config.mc.Max_VRAM_MB) * 1024.0 * 1024.0;
379
380 const int calculated_paths = static_cast<int>(gpu_budget_bytes / bytes_per_sde_path);
381
382 // Integer division to get multiples of 32 (1 warps or wavefronts)
383 // First, it ensures that every single GPU warp launched is packed with active execution threads (warp is 100% full)
384 // Second, it aligns data boundaries with the physical architecture of the GPU chip
385 // (data are read by 32 blocks, so you just make single memory transaction to read data for a single warp)
386 return (calculated_paths / 32) * 32;
387
388#else
389 double cpu_budget_bytes = static_cast<double>(config.mc.Max_CPU_RAM_MB) * 1024.0 * 1024.0;
390
391 // If you need to allocate a master matrix for backward paths on cpu, allocate this first
392 if (config.options.opt_type == KI::OptType::American) {
393 const double master_matrix_bytes = static_cast<double>(config.mc.N_Paths) * bytes_per_sde_path;
394 cpu_budget_bytes -= master_matrix_bytes;
395
396 // If the Master Matrix alone eats the entire budget, return 0.
397 // The Low-Bound Safety check in allocate_batch_memory() will catch this and throw cleanly.
398 if (cpu_budget_bytes <= 0.0) {
399 throw std::runtime_error(
400 "[Memory Capacity Error] The Longstaff-Schwartz algorithm requires CPU RAM "
401 "that, summed to the batches memory used for the forward computation, exceeds your assigned Max_CPU_RAM_MB budget.\n"
402 " -> Master Matrix RAM : " + std::to_string(master_matrix_bytes) + " MB\n"
403 " -> Host Batch Buffers : " + std::to_string(cpu_budget_bytes) + " MB\n"
404 " -> Config Budget : " + std::to_string(config.mc.Max_CPU_RAM_MB) + " MB\n"
405 "Possible Actions: \n"
406 "\t 1) Increase 'Max_CPU_RAM_MB' in your config,\n"
407 "\t 2) Reduce the the batch size (explicitly assign a moderate number of sims per batch),\n"
408 "\t 3) Reduce the number of paths/time steps.\n"
409 );
410 }
411 }
412
413 return static_cast<int>(cpu_budget_bytes / bytes_per_sde_path); //Integer division
414
415#endif
416 }
