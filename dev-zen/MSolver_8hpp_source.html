QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

90 // 2. The Execution Operator (Better than KOKKOS_LAMBDA)
98 KOKKOS_INLINE_FUNCTION
99 void operator()(const int n_p) const {
100 // Get the random generator for given threat (generator automatically freed when goes out of scope)
101 ScopedRNG scoped_rng(rng_pool);
102 auto &rn_generator = scoped_rng.return_unique_rng_state(); // create a temporary state in thread's register
103
104 // Initialize Loop Variables
105 KT::Real S = S0;
106 KT::Real v = v0;
107 PayoffTracker<OptType, OptRight> PTracker(S); // Trivially copiable object not involving pointers
108
109 // Evolve in time the current path
110 for (int n_t = 0; n_t < n_t_steps; ++n_t) {
111 // Get new state
112 auto [next_S, next_v] = Scheme.evolve_step(S, v, rn_generator);
113 S = next_S;
114 v = next_v;
115
116 // Store Price if necessary
117 if constexpr (StorePaths == true) {
118 local_paths_batch_view(n_p, n_t) = S; // Global write to GPU's VRAM
119 }
120
121 // Track Price
122 PTracker.track_current_price(S);
123 }
124
125 // Evaluate Payoff using the tracker state
126 local_payoff_batch_view(n_p) = PTracker.evaluate_final_payoff(S,
127 Strike,
128 BarrierPrice,
129 n_t_steps);
130 }
131 };

195 void execute_batch(const int n_active_sims_in_batch, PathsMCBatchMem &BatchMem, const RNGManager &RNGen) const {
196 // Lambda for setting up and launching the markovian kernel
197 auto launch_markovian_kernel = [&](auto store_paths_tag) {
198 // 0. Decide weather to store the paths or not
199 // Note: decltype() -> operator that asks the compiler: What is the type of this expression?
200 static_assert(std::is_same_v<decltype(store_paths_tag), std::bool_constant<true> > ||
201 std::is_same_v<decltype(store_paths_tag), std::bool_constant<false> >,
202 "Kernel must be launched with a std::bool_constant to chose if to store paths or not!");
203 constexpr bool StorePaths = decltype(store_paths_tag)::value;
204
205 // 1. Package data from the subsystems into the execution functor
206 // Note: the random-pull is the same for all batches, it does not have to be reinitialized !
207 MarkovianIntegrationKernel<ModelPolicy, SchemePolicy, OptType, OptRight, StorePaths> kernel{
208 Scheme,
209 BatchMem.d_batch_view,
210 BatchMem.d_payoffs,
211 RNGen.get_global_rng_pool(),
212 config.time.N_time_steps,
213 config.market.S0,
214 config.market.v0,
215 config.options.StrikePrice,
216 config.options.BarrierPrice
217 };
218
219 // 2. Safely isolate the parallel launch boundary inside the class
220 Kokkos::parallel_for("Evolve_SDEs_in_given_batch", n_active_sims_in_batch, kernel);
221 }; // End of lambda
222
223 // Dispatch Kernel based on runtime condition
224 // NOTE : bridge run time data to compile-time types
225 if (config.requires_paths_allocated()) {
226 launch_markovian_kernel(std::bool_constant<true>{});
227 } else {
228 launch_markovian_kernel(std::bool_constant<false>{});
229 }
230
231 Kokkos::fence();
232 }
233
234 private:
235 // Reference is completely safe on the Host side!
236 const KC::UInputs &config;
237 SDEScheme<ModelPolicy, SchemePolicy> Scheme;
239
244 inline bool must_store_paths() const {
245 const auto OType = config.options.opt_type;
246 const bool requires_full_paths_matrix = (OType == KI::OptType::American);
247
248 // Store if the paths to save the outputs if given or if the option evaluation requires all paths matrix for backward algorithm
249 return !config.output.filename_paths_out.empty() || requires_full_paths_matrix;
250 }
