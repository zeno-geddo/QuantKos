45 inline double get_exact_solution(const quantkos::Config::UInputs &conf) {
46 // Expected: S_T = S_0 * exp((r - q) * T)
47 const double expected_S_T = conf.market.S0 * std::exp(
48 (conf.market.r - conf.market.q) * conf.time.t_end);
49 return expected_S_T;
50 }

113 inline bool run_test() {
114 // NOTE : When no randomness, the stock should grow purely by the deterministic drift: S_T = S_0 e^{(r-q)T}.
115
116 std::string_view indent{" "};
117 std::cout << "\n\n" << indent << "====================================================================\n"
118 << indent << " TEST 3 : Zero Variance Forward Growth Check \n"
119 << indent << "====================================================================\n";
120 std::cout << indent << "[ RUN ] Zero Variance Forward Growth Check\n";
121
122 bool overall_test_passed = true;
123
124 // STEP 1: Get exact solution
125 auto config = getDefaultConfig();
126 config.validate();
127 const double expected_S_T = get_exact_solution(config);
128 std::cout << indent << "[ INFO ] Expected Final Price : " << expected_S_T << "\n";
129
130
131 // STEP 2 : Run test for each model/scheme
132 const auto models_to_test = KTU::get_models_to_test();
133 for (const auto &[fst, snd]: models_to_test) {
134 // STEP 2a : Set config for numerical solver
135 config.model.id_model = fst;
136 config.scheme.id_scheme = snd;
137 std::string name_out_file = "Test_" + KI::enum_to_string(config.model.id_model) + "_"
138 + KI::enum_to_string(config.scheme.id_scheme) + ".paths";
139 config.output.filename_paths_out = name_out_file;
140 config.validate();
141 config.print_summary();
142
143 // Check and print params
144 std::cout << indent << "[ INFO ] MODEL : " << KI::enum_to_string(config.model.id_model) << "\n";
145 std::cout << indent << "[ INFO ] SCHEME : " << KI::enum_to_string(config.scheme.id_scheme) <<
146 "\n";
147
148 std::cout << indent << ">>> Calling the solver ...\n";
149 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
150
151 // STEP 2b :Run model (keep within a dedicated the scope to be sure that dispatcher is destructed correctly)
152 {
153 auto MCDisp = KE::MCDispatcher(config);
154 MCDisp.launch_montecarlo();
155 }
156
157 //STEP 2c : Read the data back from disks (this also tests that writing and loading works correclty)
158 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
159 std::cout << indent << ">>> Go back to the tester ...\n";
160 KB::BinReader binReader(config);
161 std::vector<KT::Real> simulated_prices = binReader.read_prices_at_target_time(config.time.t_end);
162
163 // Check if last Price at T_Final is correct
164 const double epsilon = KT::is_real_using_single_precision() ? 5e-3 : 1e-8;
165 for (size_t i = 0; i < simulated_prices.size(); ++i) {
166 // Check if you have a nan
167 if (!std::isfinite(simulated_prices[i])) {
168 std::cout << indent << "[ FAILED ] Path " << i
169 << " produced a non-finite value: " << simulated_prices[i] << "\n";
170 std::cerr << indent << "[ FAILED ] Path " << i
171 << " produced a non-finite value: " << simulated_prices[i] << "\n";
172 overall_test_passed = false;
173 break;
174 }
175
176 // We use 1e-5 to account for floating point drift during 365 compounded steps
177 if (std::abs(simulated_prices[i] - expected_S_T) > epsilon) {
178 std::cout << indent << "[ FAILED ] Path " << i << " deviated! Expected: "
179 << expected_S_T << ", Got: " << simulated_prices[i] << "\n";
180 std::cerr << indent << "[ FAILED ] Path " << i << " deviated! Expected: "
181 << expected_S_T << ", Got: " << simulated_prices[i] << "\n";
182 overall_test_passed = false;
183 break;
184 }
185 std::cout << indent << "[ PASSED ] Path " << i <<
186 " (Simulated : " << simulated_prices[i] << ", Expected : " << expected_S_T << ")\n";
187 }
188
189 // Clean up out binary file generated (all the output folder folder)
190 std::filesystem::path out_dir_path = config.output.out_dir;
191 if (std::filesystem::exists(out_dir_path)) {
192 std::filesystem::remove_all(out_dir_path);
193 }
194 }
195
196 // STEP 3 : Check if the tests passed.
197 if (overall_test_passed) {
198 std::cout << indent << "[ PASSED ] All terminal prices match the theoretical drift.\n";
199 } else {
200 std::cout << indent << "[ FAILED ] Terminal prices does not match the theoretical drift.\n";
201 }
202 std::cout << indent << "====================================================================\n" << std::endl;
203 return overall_test_passed;
204 }
