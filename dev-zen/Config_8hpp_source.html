QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

172 struct Heston {

174 Real theta = 0.04;
175 Real sigma = 0.3;

185 struct Bates : public Heston {
186 Real lambda_J = 0.1;
187 Real mu_J = -0.1;
188 Real sigma_J = 0.15;
189 } bates;
190
195 void validate() const {
196 if (id_model == KI::MathModel::Heston) {
197 validate_heston_params(heston);
198 } else if (id_model == KI::MathModel::Bates) {
199 validate_bates_params(bates);
200 }
201 }
202
209 void print(const MarketConfig &market, std::string_view indent = "") const {
210 // Assemble the equations
211 std::string str_math_model = "";
212 std::string ind(indent);
213
214 if (id_model == KI::MathModel::Heston) {
215 str_math_model = ind + "\t\t\t dSt = (r - q)·St·dt + √(vt)·St·dW1,t\n" +
216 ind + "\t\t\t dvt = κ·(θ - vt)·dt + σ·√(vt)·dW2,t\n" +
217 ind + "\t\t\t where : E[dW1,t·dW2,t] = ρ·dt\n";
218 } else if (id_model == KI::MathModel::Bates) {
219 // dNt = Merton jump process
220 str_math_model = ind + "\t\t dSt = (r - q - λ·κ_J)·St·dt + √(vt)·St·dW1,t + St⁻·(J - 1)·dNt\n" +
221 ind + "\t\t dvt = κ·(θ - vt)·dt + σ·√(vt)·dW2,t\n" +
222 ind + "\t\t where :\n" + // λ·κJ = Martingale compensator
223 ind + "\t\t ln(J) ~ N(μ_J, σ_J²),\n" +
224 // log normal jump size (J = Realized Random jump size)
225 ind + "\t\t κ_J = exp(μ_J + 0.5·σ_J²) - 1,\n" +
226 // expected percentage change in the asset price caused by a single jump
227 ind + "\t\t E[dW1,t·dW2,t] = ρ·dt\n";
228 }
229
230 if (id_model == KI::MathModel::Heston || id_model == KI::MathModel::Bates) {
231 // Handle parameter reference mapping cleanly based on active execution mode (upcast if needed)
232 const Heston &base_params = (id_model == KI::MathModel::Bates)
233 ? static_cast<const Heston &>(bates)
234 : heston;
235
236 std::cout << indent << " [" << KK::Model << "]\n"
237 << indent << " ID Model : " << id_model << "\n"
238 << indent << " --------------------------------------------------------\n"
239 << str_math_model
240 << indent << " --------------------------------------------------------\n"
241 << indent << " Core Parameters :\n"
242 << indent << " Risk-free interest rate (" << KK::MarketParams::r <<
243 ") : " << market.r << "\n"
244 << indent << " Continuous dividend yield (" << KK::MarketParams::q <<
245 ") : " << market.q << "\n"
246 << indent << " Mean reversion speed of the variance (" << KK::MathModelParams::k <<
247 ") : " << base_params.k << "\n"
248 << indent << " Mean reversion level of the variance (" << KK::MathModelParams::theta <<
249 ") : " << base_params.theta << "\n"
250 << indent << " Volatility of the variance (" << KK::MathModelParams::sigma <<
251 ") : " << base_params.sigma << "\n"
252 << indent << " Price-Variance correlation (" << KK::MathModelParams::rho <<
253 ") : " << base_params.rho << "\n";
254
255 if (2.0 * base_params.k * base_params.theta <= base_params.sigma * base_params.sigma) {
256 std::cout << indent <<
257 " Feller condition (2κθ > σ²) is violated. Variance paths may touch zero.\n";
258 }
259
260 if (id_model == KI::MathModel::Bates) {
261 std::cout << indent << " Jump Parameters:\n"
262 << indent << " Jump Intensity (λ) : " << bates.lambda_J << "\n"
263 << indent << " Mean Jump Size (μJ) : " << bates.mu_J << "\n"
264 << indent << " Jump Volatility (σJ) : " << bates.sigma_J << "\n";
265 }
266 } else {
267 //const std::string msg = "[Config Error] Model selected ('" + id_model + "') is not a valid model !";
268 throw std::runtime_error(config_err_msg(KK::Model, KK::Model,
269 "Model selected is not a valid model !"));
270 }
271 }

274
276 void validate_heston_params(const Heston &h) const {
277 if (h.k < 0.0)
278 throw std::runtime_error("[Model Config] κ must be positive.");
279 if (h.theta < 0.0)
280 throw std::runtime_error("[Model Config] θ must be positive.");
281 if (h.sigma < 0.0)
282 throw std::runtime_error("[Model Config] σ must be positive.");
283 if (h.rho < -1.0 || h.rho > 1.0)
284 throw std::runtime_error("[Model Config] ρ must fall within [-1.0, 1.0].");
285 }
286
288 void validate_bates_params(const Bates &b) const {
289 // Leverage inheritance! Validate the stoch-vol layer first.
290 validate_heston_params(b);
291
292 // Validate Jump Bounds
293 if (b.lambda_J < 0.0) {
294 throw std::runtime_error("[Model Config] Bates: Jump intensity (λ) cannot be negative.");
295 }
296 if (b.sigma_J < 0.0) {
297 throw std::runtime_error("[Model Config] Bates: Jump volatility (σJ) must be positive.");
298 }
299 }

330 struct TimeConfig {
331 Real t_end = 1.0;
332 Real inp_dt = 0.2;
333 Real dt;
334 int N_time_steps;
335
341 void validate() {
342 // Check inputs
343 if (t_end <= 0.0)
344 throw std::runtime_error(
345 config_err_msg(KK::Time, KK::TimeParams::T_End, "must be positive."));
346 if (inp_dt <= 0.0)
347 throw std::runtime_error(
348 config_err_msg(KK::Time, KK::TimeParams::DT, "must be positive."));
349 if (inp_dt > t_end)
350 throw std::runtime_error(
351 config_err_msg(KK::Time, KK::TimeParams::DT, "must be smaller that total time."));
352
353 // 1. Calculate the number of steps by rounding up (ceil)
354 N_time_steps = static_cast<int>(std::ceil(t_end / inp_dt));
355
356 // 2. Uniformly distribute the time steps to fit t_end exactly
357 dt = t_end / N_time_steps;
358
359 // 3. Inform/Warn the user if their input was altered
360 // (Using a small epsilon for floating-point comparison safety)
361 if (std::abs(dt - inp_dt) > 1e-12) {
362 std::cout << "[Warning] Time step dt (" << dt
363 << "s) does not evenly divide End Time (" << t_end << ").\n"
364 << " Automatically adjusted dt to " << dt
365 << "across " << N_time_steps << " uniform steps.\n" << std::endl;
366 }
367 }
368
373 void print(std::string_view indent = "") const {
374 std::cout << indent << " [" << KK::Time << "]\n"
375 << indent << " End Time (" << KK::TimeParams::T_End << ") : " << t_end <<
376 "\n"
377 << indent << " User Input Time Step (" << KK::TimeParams::Inp_DT << ") : " << inp_dt <<
378 "\n"
379 << indent << " Actual Time Step (" << KK::TimeParams::DT << ") : " << dt << "\n"
380 << indent << " Number Actual Time Steps (" << KK::TimeParams::N_TSteps << ") : " <<
381 N_time_steps << "\n";
382 }
383 };

405 void validate() const {

412 "and 0 if you want to let the machine handle this."));
413 if (batch_size > N_Paths)
414 throw std::runtime_error(config_err_msg(
415 KK::MC, KK::MCParams::Batch_Size,
416 "The batch size must be smaller that the notal number of realizations."));
417 if (rng_seed == 0) {
418 throw std::runtime_error(config_err_msg(KK::MC, KK::MCParams::RNG_Seed,
419 "The random number seed cannot be 0."));
420 }
421 if (rng_seed >= 18446744073709551615ULL) {
422 config_err_msg(KK::MC, KK::MCParams::RNG_Seed,
423 "[Warning] Seed is at maximum uint64 limit. Did you pass a negative number?");
424 }
425 if (Max_VRAM_MB <= 0)
426 throw std::runtime_error(
427 config_err_msg(KK::MC, KK::MCParams::Max_VRAM_MB, "must be a positive integer."));
428 if (Max_CPU_RAM_MB <= 0)
