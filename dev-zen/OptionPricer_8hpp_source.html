It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

142 void accumulate_batch_payoffs(const DevBatchPayoffView &d_payoffs,
143 const int curr_batch_size) {
144 // STEP 1 : Get Batch Sum and ITM count on device
145 PayoffsBatchStats batch_stats;
146 const auto eps = eps_payoff;
147 Kokkos::parallel_reduce("Batch_Sum_And_ITM_Pass", curr_batch_size,
148 KOKKOS_LAMBDA(const int i, PayoffsBatchStats &l_stats) {
149 const double payoff = static_cast<double>(d_payoffs(i));
150 l_stats.sum += payoff; // Accumulate sum
151 if (payoff > eps) l_stats.itm += 1; // Accumulate ITM count
152 }, batch_stats);
153 const double payoff_batch_sum = batch_stats.sum;
154 const size_t batch_itm_count = batch_stats.itm;
155 const double batch_mean = payoff_batch_sum / curr_batch_size;
156
157
158 // STEP 2: Get Batch M2 (Sum of squared differences) on device---
159 double payoff_batch_sum_diff2 = 0.0;
160 Kokkos::parallel_reduce("Batch_M2_Pass", curr_batch_size,
161 KOKKOS_LAMBDA(const int i, double &sum_diff2) {
162 const double diff = static_cast<double>(d_payoffs(i)) - batch_mean;
163 sum_diff2 += (diff * diff);
164 }, payoff_batch_sum_diff2);
165
166
167 // ====================================================================
168 // --- CPU MERGE: Mean + Chan's Exact Variance ---
169 // ====================================================================
170 const size_t new_total_paths = n_payoffs_processed + curr_batch_size;
171
172 // 1. Calculate delta using the OLD global_mean (Required for payoff_sum_diff2)
173 const double delta = batch_mean - payoff_mean;
174
175 // 2. Weighted Average to update the mean
176 payoff_mean = (payoff_mean * static_cast<double>(n_payoffs_processed) +
177 batch_mean * static_cast<double>(curr_batch_size)
178 ) / static_cast<double>(new_total_paths);
179
180 // 3. Chan's Formula to update sum of diff squared (needed for variance merging)
181 // Note : M2 Chang's parallel algo : https://en.wikipedia.org/wiki/Algorithms_for_calculating_variance
182 payoff_M2_Chang += payoff_batch_sum_diff2 +
183 (delta * delta) *
184 (static_cast<double>(n_payoffs_processed * curr_batch_size)
185 / static_cast<double>(new_total_paths));
186
187 // Update trackers
188 n_payoffs_processed = new_total_paths;
189 itm_count += batch_itm_count;
190
191
192 // STEP 4 : OPTIONAL: Store arrays ONLY if distribution analysis is active ---
193 if (config.mc.analyze_risk_neutral_payoff_distribution) {
194 // 1. Destination subview (Where to put it in the global array)
195 const auto dest_slice = std::make_pair(current_path_idx, current_path_idx + curr_batch_size);
196 const auto destination_subview = Kokkos::subview(all_payoffs, dest_slice);
197
198 // 2. Source subview (Only copy the valid computed paths from this batch buffer!)
199 const auto src_slice = std::make_pair(0, curr_batch_size);
200 const auto source_subview = Kokkos::subview(d_payoffs, src_slice);
201
202 // 3. Perform device-to-device copy with matching extents
203 Kokkos::deep_copy(destination_subview, source_subview);
204 }
205 }

216 void evaluate_option_price() {
217 const auto N = static_cast<double>(config.mc.N_Paths);
218 if (N <= 0) {
219 throw std::runtime_error("Cannot compute metrics: Invalid number of paths.");
220 }
221
222 // 1. Compute discount factor (do not apply to backwards options since already applied)
223 double discount_factor = KT::real_one;
224 if (config.options.opt_type != KI::OptType::American) {
225 const auto r = static_cast<double>(config.market.r);
226 const auto t = static_cast<double>(config.time.t_end);
227 discount_factor = std::exp(-r * t); // Safely compute in double precision
228 }
229
230 // 2. Compute Unbiased sample variance: M2 / (N - 1)
231 // (We add a small ternary check to prevent division by zero if N == 1)
232 const double payoff_variance = (N > 1.0) ? (payoff_M2_Chang / (N - 1.0)) : 0.0;
233
234 // 3. Apply Discounting and scaling to get the Option Price etc.
235 // Note: standard_error : how confident we are that our calculated average is the true option price
236 const double effective_scale = discount_factor * get_scaling_factor();
237 metrics.option_price = payoff_mean * effective_scale;
238 metrics.standard_error = std::sqrt(payoff_variance / N) * effective_scale;
239 metrics.prob_itm = static_cast<double>(itm_count) / N;
240
241
242 // 4. Analyze shape of option price distribution ONLY if requested
243 if (config.mc.analyze_risk_neutral_payoff_distribution) {
244 if (all_payoffs.empty()) {
245 // Safety check just in case memory allocation failed earlier
246 throw std::runtime_error("Distribution analysis requested, but payoffs array is empty.");
247 }
248
249 const auto start_time = std::chrono::steady_clock::now();
250
251 analyze_risk_neutral_discounted_payoff_distribution(effective_scale, N);
252
253 // Store time
254 const auto end_time = std::chrono::steady_clock::now();
255 const std::chrono::duration<double> elapsed = end_time - start_time;
256 sorting_time = elapsed.count();
257 }
258
259 // Mark as successfully computed
260 metrics.is_computed = true;
261 }
