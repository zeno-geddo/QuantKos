139 inline bool test_default_kokkos_gaussian() {
140 const std::string_view indent{" "};
141 std::cout << "\n\n" << indent << "====================================================================\n"
142 << indent << " TEST 1a : Standard Normal Distribution Check (kokkos box-muller) \n"
143 << indent << "====================================================================\n";
144
145 std::cout << indent << "[ RUN ] Standard Normal Distribution Check\n";
146
147
148 bool test_passed = true;
149
150 // Define size test
151 const int batch_size = 1'000'000;
152 const int N_time_steps = 365;
153
154 // Initialize Random number generation manager
155 const auto rng_man = initialize_rng_manager();
156
157 // Allocate Memory for the generated random number
158 auto d_z_values = Kokkos::View<KT::Real **>("device_Z_values",
159 batch_size,
160 N_time_steps);
161 auto h_z_values = Kokkos::create_mirror_view(d_z_values);
162
163 // Launch Dummy Kernel to see if we are really generating guassina values
164 const auto start_time = std::chrono::steady_clock::now();
165 execute_rng_kernel_kookkos_def(batch_size,
166 N_time_steps,
167 d_z_values,
168 rng_man.get_global_rng_pool());
169 const auto end_time = std::chrono::steady_clock::now();
170 const std::chrono::duration<double> elapsed = end_time - start_time;
171 std::cout << indent << "[ INFO ] Kernel exe time : "<< elapsed.count()<< "s \n";
172
173 // Copy results on host if not already there
174 Kokkos::deep_copy(h_z_values, d_z_values);
175
176 // Calculate Statistics
177 double sum = 0.0;
178 double sq_sum = 0.0;
179 for (int i = 0; i < batch_size; ++i) {
180 for (int j = 0; j < N_time_steps; ++j) {
181 const auto z = static_cast<double>(h_z_values(i, j));
182 sum += z;
183 sq_sum += (z * z);
184 }
185 }
186
187 const auto N_draws = static_cast<double>(batch_size * N_time_steps);
188 const double mean = sum / N_draws;
189 const double variance = (sq_sum / N_draws) - (mean * mean);
190
191 std::cout << indent << "[ INFO ] Number Samples : " << N_draws << "\n";
192 std::cout << indent << "[ INFO ] Measured Mean : " << mean << "\n";
193 std::cout << indent << "[ INFO ] Measured Variance : " << variance << "\n";
194
195 if (std::abs(mean - 0.0) > 0.005) {
196 std::cerr << indent << "[ FAILED ] RNG Mean drifted unacceptably far from 0.0!\n";
197 test_passed = false;
198 }
199
200 if (std::abs(variance - 1.0) > 0.01) {
201 std::cerr << indent << "[ FAILED ] RNG Variance drifted unacceptably far from 1.0!\n";
202 test_passed = false;
203 }
204
205 if (test_passed) {
206 std::cout << indent << "[ PASSED ] Standard Normal Distribution Check \n";
207 }
208
209
210 std::cout << indent << "====================================================================\n" << std::endl;
211
212 return test_passed;
213 }

236 struct CustomGaussianRNGTestKernel {
237 int n_t_steps;
238 Kokkos::View<KT::Real **> dummy_path_view;
239 KE::RNGManager::GlobalRNGPool rng_pool;
240 DummyCustomGaussianSDEScheme Scheme;
241
242 KOKKOS_INLINE_FUNCTION
243 void operator()(const int n_p) const {
244 quantkos::Engine::ScopedRNG scoped_rng(rng_pool);
245 auto &rn_generator = scoped_rng.return_unique_rng_state();
246
247 // Advance by 2 because our functor generates TWO numbers at once
248 for (int i = 0; i < n_t_steps; i += 2) {
249 const KE::RVPair z_pair = Scheme.evolve_step(rn_generator);
250
251 // Always write Z1
252 dummy_path_view(n_p, i) = z_pair.Z1;
253
254 // Only write Z2 if we haven't exceeded the time steps (e.g., if n_t_steps is odd like 365)
255 if (i + 1 < n_t_steps) {
256 dummy_path_view(n_p, i + 1) = z_pair.Z2;
257 }
258 }
259 }
260 };
