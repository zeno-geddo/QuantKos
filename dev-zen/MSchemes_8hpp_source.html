It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

87 template<>
88 struct SDEScheme<KI::MathModel::Heston, KI::NumScheme::Euler> {
89 // Weak Convergence 1., Strong convergence 0.5
90 // Precomputed constant scalar invariants (Computed once on CPU Host)
91 KT::Real one_minus_k_dt;
92 KT::Real k_theta_dt;
93 KT::Real sigma_sqrt_dt;
94 KT::Real r_minus_q_dt;
95 KT::Real half_dt;
96 KT::Real sqrt_dt;
97 KT::Real rho;
98 KT::Real rho_complement;
99
100 // Constructor
102 explicit SDEScheme(const KC::UInputs &config) {
103 const KT::Real r = config.market.r;
104 const KT::Real q = config.market.q;
105 const KT::Real k = config.model.heston.k;
106 const KT::Real theta = config.model.heston.theta;
107 const KT::Real sigma = config.model.heston.sigma;
108 rho = config.model.heston.rho;
109 const KT::Real dt = config.time.dt;
110
111 // Compute roots once on the host so the GPU doesn't have to
112 sqrt_dt = Kokkos::sqrt(dt);
113 rho_complement = Kokkos::sqrt(KT::real_one - rho * rho);
114
115 // PreCompute variance factors
116 one_minus_k_dt = KT::real_one - k * dt;
117 k_theta_dt = k * theta * dt;
118 sigma_sqrt_dt = sigma * sqrt_dt;
119
120 // PreCompute price factors
121 r_minus_q_dt = (r - q) * dt;
122 half_dt = KT::real_05 * dt; // Avoid division on GPU since it is a slow operation
123 }

132 template<typename RNGeneratorType>
133 KOKKOS_INLINE_FUNCTION SDEState evolve_step(const KT::Real S_n,
134 const KT::Real v_n,
135 RNGeneratorType &local_rn_generator) const {
136 // Get Random Normal Variables
137 //const KT::Real Z_1 = static_cast<KT::Real>(local_rn_generator.normal());
138 //const KT::Real Z_2 = static_cast<KT::Real>(local_rn_generator.normal());
139 const auto [Z_1, Z_2] = NormalPair<RNGeneratorType>{}(local_rn_generator);
140
141 // Cholesky Decomposition for correlated Brownian Motion
142 const KT::Real Z_v = Z_1;
143 const KT::Real Z_S = rho * Z_1 + rho_complement * Z_2;
144
145 // EVOLVE ASSET (NOTE: MUST USE OLD VARIANCE TO RESPECT ITO INTEGRAL, REQUIRING WITH THE VAL AT BEGINNIG OF TIME SETP)
146 // Mathematically: (r - q)dt - (v_n * 0.5 * dt) + (sqrt(v_n) * sqrt(dt)) * Z_S
147 const KT::Real sqrt_v_n = Kokkos::sqrt(v_n);
148 const KT::Real exponent = r_minus_q_dt - (v_n * half_dt) + (sqrt_v_n * sqrt_dt) * Z_S;
149 const KT::Real S_np1 = S_n * Kokkos::exp(exponent); // Special Function Unit (SFU) Call (slow)
150
151 // EVOLVE & TRUNCATE VARIANCE (Fused Multiply-Add (FMA) optimized variance step)
152 KT::Real v_np1 = v_n * one_minus_k_dt + k_theta_dt + (sigma_sqrt_dt * sqrt_v_n) * Z_v;
153 v_np1 = Kokkos::fmax(v_np1, KT::real_zero);
154
155 return SDEState{.S = S_np1, .v = v_np1};
156 }
157 };

227 KOKKOS_INLINE_FUNCTION SDEState evolve_step(const KT::Real S_n,
228 const KT::Real v_n,
229 RNGeneratorType &local_rn_generator) const {
230 // Get Random Normal Variables
231 //const KT::Real Z_1 = static_cast<KT::Real>(local_rn_generator.normal());
232 //const KT::Real Z_2 = static_cast<KT::Real>(local_rn_generator.normal());
233 const auto [Z_1, Z_2] = NormalPair<RNGeneratorType>{}(local_rn_generator);
234
235 // Cholesky Decomposition for correlated Brownian Motion
236 const KT::Real Z_v = Z_1;
237 const KT::Real Z_S = rho * Z_1 + rho_complement * Z_2;
238
239
240 // EVOLVE EVOLUTION (Evaluated at t_n to respect Ito calculus)
241 const KT::Real sqrt_v_n = Kokkos::sqrt(v_n);
242 const KT::Real exponent = r_minus_q_dt - (v_n * half_dt) + (sqrt_v_n * sqrt_dt) * Z_S;
243 const KT::Real S_np1 = S_n * Kokkos::exp(exponent);
244
245 // EVOLVE & TRUNCATE VARIANCE VARIANCE EVOLVE: Equation 7.26 (Algebraic Implicit Milstein)
246 const KT::Real milstein_correction = quarter_sigma_sq_dt * (Z_v * Z_v - KT::real_one);
247 KT::Real v_np1 = implicit_denominator_v *
248 (v_n + k_theta_dt + (sigma_sqrt_dt * sqrt_v_n) * Z_v + milstein_correction);
249 v_np1 = Kokkos::fmax(v_np1, KT::real_zero);
250 return SDEState{.S = S_np1, .v = v_np1};
251 }

274 template<>
275 struct SDEScheme<KI::MathModel::Heston, KI::NumScheme::AndersonQE> {
276 // Precomputed constant scalar invariants
277 KT::Real r_minus_q_dt;
278 KT::Real half_dt;
279 KT::Real sqrt_dt;
280 KT::Real rho;
281 KT::Real rho_complement;
282
283 // QE-Specific eps security values
284 KT::Real eps_param = KT::is_real_using_single_precision() ? 1e-6f : 1e-12;
285 KT::Real eps_psi = KT::is_real_using_single_precision() ? 1e-6f : 1e-12;
286
287 // QE-Specific Structural Constants
288 KT::Real psi_c = KT::real_1p5; // 1.5 as in Anderson
289 KT::Real gamma1 = KT::real_05; // Central Predictor-Corrector
290 KT::Real gamma2 = KT::real_05;
291 KT::Real inv_sqrt_2; // For the Error Function (Normal CDF)
292
293 // QE-Specific invariants
294 KT::Real exp_minus_k_dt;
295 KT::Real theta_m_factor; // theta * (1 - e^{-k dt})
296 KT::Real s2_factor_1; // (sigma^2 * e^{-k dt} / k) * (1 - e^{-k dt})
297 KT::Real s2_factor_2; // (theta * sigma^2 / (2k)) * (1 - e^{-k dt})^2
298 KT::Real K1, K2, K3, K4, A;
299
301 explicit SDEScheme(const KC::UInputs &config) {
302 const KT::Real r = config.market.r;
303 const KT::Real q = config.market.q;
304 const KT::Real k = config.model.heston.k;
305 const KT::Real theta = config.model.heston.theta;
306 const KT::Real sigma = config.model.heston.sigma;
307 rho = config.model.heston.rho;
308 const KT::Real dt = config.time.dt;
309
310 // Base invariants
311 sqrt_dt = std::sqrt(dt);
312 r_minus_q_dt = (r - q) * dt;
313 half_dt = KT::real_05 * dt;
314
315 // Anderson QE invariants
316 inv_sqrt_2 = static_cast<KT::Real>(1.0 / std::sqrt(2.0));
317 exp_minus_k_dt = std::exp(-k * dt);
318
319 // Handle case k = 0
320 if (k < eps_param) {
321 // Catch case when there is not drift
322 // Compute them with theorem de l'hopital
323 theta_m_factor = KT::real_zero;
324 s2_factor_1 = sigma * sigma * dt;
325 s2_factor_2 = KT::real_zero;
326 } else {
327 const KT::Real one_minus_exp = KT::real_one - exp_minus_k_dt;
328 theta_m_factor = theta * one_minus_exp; // in Eq 7.41, F. Rouah
329 const KT::Real sig_sq = sigma * sigma; // in Eq 7.41, F. Rouah
330 s2_factor_1 = (sig_sq * exp_minus_k_dt / k) * one_minus_exp; // in Eq, 7.41 F. Rouah
331 s2_factor_2 = (theta * sig_sq / (KT::real_two * k)) * (one_minus_exp * one_minus_exp);
332 }
333
334 // Handle case sigma=0
335 KT::Real rho_div_simga;
336 if (sigma < eps_param) {
337 // To handle Black Scholes degeneration
338 rho_div_simga = KT::real_zero;
339 rho = KT::real_zero; // If sigma is near zero, correlation is mathematically meaningless
340 } else {
341 rho_div_simga = rho / sigma; // after Eq 7.47, F. Rouah
342 }
343 const KT::Real one_m_rho2 = KT::real_one - rho * rho; // after Eq 7.47, F. Rouah
344 const KT::Real K12_fact = (k * rho_div_simga - KT::real_05); // after Eq 7.47, F. Rouah
345 K1 = dt * gamma1 * K12_fact - rho_div_simga; // after Eq 7.47, F. Rouah
346 K2 = dt * gamma2 * K12_fact + rho_div_simga; // after Eq 7.47, F. Rouah
347 K3 = dt * gamma1 * one_m_rho2; // after Eq 7.47, F. Rouah
348 K4 = dt * gamma2 * one_m_rho2; // after Eq 7.47, F. Rouah
349 // Martingale Constants precomputed for the GPU
350 A = K2 + K4 * KT::real_05; // Terms for matingale correction K0
351 }

360 template<typename RNGeneratorType>
361 KOKKOS_INLINE_FUNCTION SDEState evolve_step(const KT::Real S_n,
362 const KT::Real v_n,
363 RNGeneratorType &local_rn_generator) const {
364 // Generate noise
365 //const KT::Real Z_V = static_cast<KT::Real>(local_rn_generator.normal());
366 //const KT::Real Z_indep = static_cast<KT::Real>(local_rn_generator.normal());
367 const auto [Z_V, Z_indep] = NormalPair<RNGeneratorType>{}(local_rn_generator);
368
369 // Determine Distribution Shape (Psi)
370 // (Compute Conditional Mean (m) and Variance (s^2) of V(t+dt))
371 const KT::Real m = v_n * exp_minus_k_dt + theta_m_factor; // (Eq. 7.41, F. Rouah)
372 const KT::Real s2 = v_n * s2_factor_1 + s2_factor_2; // (Eq. 7.41, F. Rouah)
373 // const KT::Real psi = s2 / (m * m); // (After Eq. 7.42, F. Rouah)
374
375 // ====================================================================
376 // DIVERGENCE CONTROL BLOCK (for variance computation)
377 // ====================================================================
378 KT::Real v_np1;
379 KT::Real M; // Terms for martingale correction used to compute K0
380 if (m <= eps_param) {
381 // REGIME 0a: Case when m=0 since vn=0 and theta=0
382 v_np1 = KT::real_zero;
383 M = KT::real_one;
384 } else {
385 const KT::Real psi = s2 / (m * m); // (After Eq. 7.42, F. Rouah)
386 if (psi <= eps_psi) {
387 // REGIME 0b: Deterministic Limit (Black-Scholes Degeneration)
388 // NOte: this first if will not cause divergence because all threads will go here if using BS
389 // If psi is zero, variance is deterministic. Bypass the division-by-zero.
390 v_np1 = m;
391 M = KT::real_one; // The MGF of a deterministic constant is exactly 1.0
392 } else if (psi <= psi_c) {
393 // REGIME 1: Quadratic (High Variance)
394 const KT::Real inv_psi = KT::real_one / psi;
395 const KT::Real b2 = KT::real_two * inv_psi - KT::real_one +
396 Kokkos::sqrt(KT::real_two * inv_psi *
397 (KT::real_two * inv_psi - KT::real_one)
398 ); // Eq. 7.42, F. Rouah
399 const KT::Real a = m / (KT::real_one + b2); // Eq. 7.42, F. Rouah
400 const KT::Real b_plus_Zv = Kokkos::sqrt(b2) + Z_V; // Eq. 7.37, F. Rouah
401 v_np1 = a * b_plus_Zv * b_plus_Zv; // Eq. 7.37, F. Rouah
402 // Term for Martingale correction
403 const KT::Real safe_A = Kokkos::fmin(A, (KT::real_05 / a) - eps_param);
404 const KT::Real term_Aa = KT::real_one - KT::real_two * safe_A * a;
405 M = Kokkos::exp((safe_A * b2 * a) / term_Aa) / Kokkos::sqrt(term_Aa); // Eq. 7.50, F. Rouah
406 } else {
407 // REGIME 2: Exponential (Low Variance)
408 const KT::Real p = (psi - KT::real_one) / (psi + KT::real_one); // Eq. 7.43, F. Rouah
409 const KT::Real beta = (KT::real_one - p) / m; // Eq. 7.43, F. Rouah
410
411 // Recover the Uniform variable U_V from Z_V : U_V = CDF(Z_V).
412 const KT::Real U_V = KT::real_05 * (KT::real_one + Kokkos::erf(Z_V * inv_sqrt_2));
413 // Eq. 7.40, F. Rouah
414 // Update v
415 if (U_V <= p) {
416 v_np1 = KT::real_zero; // Eq. 7.40, F. Rouah
417 } else {
418 // Protect against log singularity at U_V == 1.0
419 const KT::Real safe_U = Kokkos::fmin(U_V, KT::real_one - eps_param);
420 v_np1 = (Kokkos::log((KT::real_one - p) / (KT::real_one - safe_U))) / beta;
421 // Eq. 7.40, F. Rouah
422 }
423 // Term for Martingale correction
424 const KT::Real safe_A = Kokkos::fmin(A, beta - eps_param);
425 M = p + (beta * (KT::real_one - p)) / (beta - safe_A); // Eq. 7.51, F. Rouah
426 }
427 }
428
429 // ====================================================================
430 // ASSET EVOLUTION (Predictor-Corrector Integration)
431 // ====================================================================
432 const KT::Real K0 = -Kokkos::log(M) - (K1 + K3 * KT::real_05) * v_n; // Eq. 7.49, F. Rouah (4Martingale)
433 const KT::Real integrated_sigma = (K3 * v_n + K4 * v_np1); // Eq. 7.48, F. Rouah
434 const KT::Real exponent = r_minus_q_dt + K0 +
435 (K1 * v_n + K2 * v_np1) +
436 (Kokkos::sqrt(integrated_sigma) * Z_indep); // Eq. 7.48, F. Rouah
437 const KT::Real S_np1 = S_n * Kokkos::exp(exponent); // Eq. 7.48, F. Rouah
438
439 return SDEState{.S = S_np1, .v = v_np1};
440 }
441 };

466 template<KI::NumScheme SchemePolicy>
467 struct SDEScheme<KI::MathModel::Bates, SchemePolicy> {
468 // 1. Composition: The underlying Heston Engine
469 SDEScheme<KI::MathModel::Heston, SchemePolicy> heston_core;
470
471 // 2. Bates-Specific Invariants
472 KT::Real lambda_dt;
473 KT::Real exp_minus_lambda_dt;
474 KT::Real mu_J;
475 KT::Real sigma_J;
476
478 explicit SDEScheme(const KC::UInputs &config)
479 : heston_core(create_spoofed_config_from_original(config)) {
480 // Precompute jump parameters for the GPU
481 lambda_dt = config.model.bates.lambda_J * config.time.dt;
482 exp_minus_lambda_dt = std::exp(-lambda_dt);
483 mu_J = config.model.bates.mu_J;
484 sigma_J = config.model.bates.sigma_J;
485 }

490 template<typename RNGeneratorType>
491 KOKKOS_INLINE_FUNCTION SDEState evolve_step(const KT::Real S_n,
492 const KT::Real v_n,
493 RNGeneratorType &local_rn_generator) const {
494 // 1. Evolve the continuous part using the spoofed Heston core
495 auto [S_temp, v_next] = heston_core.evolve_step(S_n, v_n, local_rn_generator);
496
497 // 2. Poisson Draw (Draw a random integer N from a Poisson distribution with mean λΔt)
498 // Note : This represents how many times the stock jumps during this specific time step.
499 // Note : The KRUNT algorithm is used (https://math.stackexchange.com/questions/3628801/proving-knuth-s-algorithm-for-generating-a-poisson-distribution)
500 int N = -1; // Start at -1 since Krunt's loop overshoots by 1. It draws the poisson jump count
501 KT::Real p = KT::real_one;
502 do {
503 // Multiply random fractions util accumulated value drops below the target threshold
504 // The number of multiplications done maps to the exact number of jumps occurred inside the discrete interval
505 N++;
506 p *= static_cast<KT::Real>(local_rn_generator.drand());
507 } while (p >= exp_minus_lambda_dt);
508
509 // 3. Draw the jump size and update price
510 KT::Real S_next = S_temp;
511 if (N > 0) {
512 // Compute current jump size
513 // Note : The total log-jump size is \sum_{i=1}^{N}(mu_J + sigma_J*Z_{J,I}) = N*mu_J + sqrt(N)*sigma_J*Z_{J}
514 const auto real_N = static_cast<KT::Real>(N);
515 const KT::Real Z = static_cast<KT::Real>(local_rn_generator.normal());
516 const KT::Real aggregate_mu = real_N * mu_J;
517 const KT::Real aggregate_sigma = Kokkos::sqrt(real_N) * sigma_J;
518 const KT::Real jump_magnitude = Kokkos::exp(aggregate_mu + aggregate_sigma * Z);
519
520 // Update Price : multiply the continuous stock price by the exponentiated jump sum.
521 S_next *= jump_magnitude;
522 }
523 return SDEState{.S = S_next, .v = v_next};
524 }

527 // Helper function that runs exclusively on the Host during construction
533 KOKKOS_INLINE_FUNCTION static KC::UInputs create_spoofed_config_from_original(const KC::UInputs &orig_config) {
534 KC::UInputs spoofed_config = orig_config; // copy the input config
535
536 // 1. C++ Object Slicing: Copies ALL base Heston fields (k, theta, sigma, rho) in one go
537 spoofed_config.model.heston = static_cast<Config::MathModelConfig::Heston>(orig_config.model.bates);
538
539 // 2. Calculate the Bates Martingale Compensator
540 const KT::Real mu = orig_config.model.bates.mu_J;
541 const KT::Real sig2 = orig_config.model.bates.sigma_J * orig_config.model.bates.sigma_J;
542 const KT::Real kappa_J = std::exp(mu + KT::real_05 * sig2) - KT::real_one; // Theoretical expected jump mean
543
544 // 3. Spoof the continuous dividend yield: q_new = q_old + (lambda * kappa_J)
545 spoofed_config.market.q += (orig_config.model.bates.lambda_J * kappa_J);
546
547 return spoofed_config;
548 }
