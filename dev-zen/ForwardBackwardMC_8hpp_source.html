QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

127 static OptionPricer::MCOpPrices compute_option_price(const KC::UInputs &loc_config,
128 BackwardLSMMemory &Mem) {
129 // 1. Initialize Helper Classes (RNGManager guarantees CRN using loc_config seed)
130 // IMPORTANT: We force the forward solver to act like a European option (simply evaluate the forward paths).
131 RNGManager RNGen(loc_config);
132 MSolver<ModelPolicy, SchemePolicy, KI::OptType::European, OptRight> Solver(loc_config);
133 OptionPricer OPricer(loc_config);
134 KIO::OutputManager OWriter(loc_config);
135 ForwardMCProgressTracker MCTracker(loc_config, Mem.BatchMem);
136
137 // 2. Print Pre-Execution Diagnostics if needed
138 MCTracker.print_pre_execution_diagnostic();
139 OWriter.print_paths_info_planned_outputs();
140
141
142 // PHASE 1: Forward Batch Generation
143 run_forward_phase(Mem, RNGen, Solver, OWriter, MCTracker); //, called_to_compute_greeks);
144
145 // PHASE 2: Backward Induction (Longstaff-Schwartz regression)
146 // CRITICAL: Pass loc_config down so Rho uses the bumped interest rate!
147 run_backward_phase(Mem, loc_config); //, called_to_compute_greeks);
148
149 // PHASE 3: Compute Final Option Price
150 // Feed the optimized backward cashflows into the standard pricer
151 OPricer.accumulate_batch_payoffs(Mem.d_best_future_outcomes, loc_config.mc.N_Paths);
152 OPricer.evaluate_option_price();
153
154 OPricer.print_info_option_price();
155
156 return OPricer.get_option_price_data();
157 }

195 // PHASE 2: BACKWARD INDUCTION (Read from CPU RAM)
196 // ====================================================================
204 static void run_backward_phase(BackwardLSMMemory &Mem,
205 const KC::UInputs &loc_config // Reads from spoofed config!
206 ) {
207 const int nT = loc_config.time.N_time_steps;
208 const KT::Real strike_price = loc_config.options.StrikePrice;
209 const KT::Real discount_factor = Kokkos::exp(-loc_config.market.r * loc_config.time.dt);
210
211 using EngineLSM = LSM::LSMEngine<OptRight>;
212
213 std::cout << " >>> Starting Phase 2: Backward Induction (PCIe Streaming)...\n";
214 BackwardLSMProgressTracker BTracker(loc_config);
215 BTracker.start_tracking();
216
217 // Initialization (t = T-1)
218 Mem.bring_host_prices_time_slice_to_device(nT - 1);
219 EngineLSM::initialize_cashflows(Mem.d_prices_current_time,
220 Mem.d_best_future_outcomes,
221 strike_price);
222 BTracker.update_progress();
223
224 // THE BACKWARD TIME MARCH
225 for (int t = nT - 2; t > 0; --t) {
226 // A. Memory Orchestration: Stream CPU RAM -> GPU VRAM
227 Mem.bring_host_prices_time_slice_to_device(t);
228
229 // B. Math Orchestration: Regression
230 const auto ls_coeffs = EngineLSM::perform_cross_paths_regression(Mem.d_prices_current_time,
231 Mem.d_best_future_outcomes,
232 discount_factor,
233 strike_price);
234
235 // C. Math Orchestration: Early Exercise Evaluation
236 EngineLSM::update_cashflows(Mem.d_prices_current_time,
237 Mem.d_best_future_outcomes,
238 ls_coeffs,
239 discount_factor,
240 strike_price);
241 BTracker.update_progress();
242 }
243
244 // 3. FINALIZATION (t = 0)
245 EngineLSM::apply_final_discount(Mem.d_best_future_outcomes, discount_factor);
246
247 // Pull final results back to CPU (Not needed)
248 // Mem.bring_device_cash_flows_to_host();
249
250 // Stop tracking the backward phase (safe since no prints in between)
251 BTracker.finalize_tracking();
252 }
