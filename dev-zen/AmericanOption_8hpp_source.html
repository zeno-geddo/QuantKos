QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

109 inline bool run_test() {
110 const std::string_view indent = " ";
111 bool all_tests_passed = true;
112 auto config = getDefaultConfigGoodIntegrand();
113
114 // Reference values from Rouah, pag 212 (from Ikonen and Toivanen (2008))
115 const std::map<double, double> S0_putPrice_map{
116 {8., 2.0000},
117 {9., 1.107641},
118 {10., 0.520030},
119 {11., 0.213668},
120 {12., 0.082036},
121 };
122
123 for (const auto pair: S0_putPrice_map) {
124 const double S0 = pair.first;
125 const double expected_price = pair.second;
126
127 config.market.S0 = S0;
128 std::cout << "\n\n" << indent << ">>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>\n";
129 std::cout << indent << "[ INFO ] S0 : " << S0 << " ; expected price : " << expected_price << " \n";
130
131 // Run Simulation
132 std::cout << indent << ">>> Calling the solver ...\n";
133 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
134 const auto MCRes = KTU::run_simulation(config);
135
136 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
137 std::cout << indent << ">>> Go back to the tester ...\n";
138
139 // Explicitly check for NaNs and Infinities first
140 if (!std::isfinite(MCRes.OptionPrice.option_price) || !std::isfinite(MCRes.OptionPrice.standard_error)) {
141 std::cout << indent << "[ FAILED ] Simulation returned non-finite numbers (NaN or Inf)!" << std::endl;
142 std::cerr << indent << "[ FAILED ] Simulation returned non-finite numbers (NaN or Inf)!" << std::endl;
143 all_tests_passed = false;
144 continue; // Skip the rest of the checks for this strike
145 }
146
147 // Check absolute error
148 const KT::Real num_price = MCRes.OptionPrice.option_price;
149 const KT::Real abs_error = std::abs(num_price - expected_price);
150 const KT::Real rel_error = (std::abs(num_price - expected_price) / expected_price) * static_cast<KT::Real>(100.);
151
152 // Statistical Tolerance (3-Sigma Rule: 99.7% Confidence Interval)
153 const KT::Real stat_error = MCRes.OptionPrice.standard_error;
154 const KT::Real stat_tolerance = 3.0 * stat_error;
155 // Discretization Bias Tolerance (Max 2% relative error, with a tiny floor for deep OTM)
156 const KT::Real bias_tolerance = std::max(expected_price * 0.02, 0.005);
157 // Total allowed tolerance
158 const KT::Real total_tolerance = stat_tolerance + bias_tolerance;
159
160 std::cout << indent << " Ref price: " << expected_price
161 << " | Num Price: " << num_price
162 << " \n | Abs Error: " << abs_error
163 << " | Rel Error %: " << rel_error
164 << " \n | Stat Error: " << stat_error
165 << " | Bias: " << bias_tolerance
166 << " | Allowed Tol Err: " << total_tolerance << "\n";
167
168 if (abs_error > total_tolerance) {
169 std::cerr << indent << "[ FAILED ] Error exceeds composite tolerance "
170 "(statistical tolerance + bias tolerance)!" << std::endl;
171 all_tests_passed = false;
172 } else {
173 std::cout << indent << "[ PASSED ] Result is within statistical bounds." << std::endl;
174 }
175 }
176 return all_tests_passed;
177 }
