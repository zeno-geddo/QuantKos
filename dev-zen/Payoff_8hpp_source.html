QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

108 KOKKOS_INLINE_FUNCTION void track_current_price(const KT::Real S) {
109 if constexpr (OptType == KI::OptType::Asian) {
110 S_sum += static_cast<double>(S); // Safe double-precision accumulation
111 }
112 // Group all options that care about the path CEILING
113 else if constexpr (
114 // Barrier types
115 OptType == KI::OptType::BarrierUpAndOut ||
116 OptType == KI::OptType::BarrierUpAndIn ||
117 // Lookback types
118 (OptType == KI::OptType::LookbackFixedStrike && OptRight == KI::OptRight::Call) ||
119 (OptType == KI::OptType::LookbackFloatingStrike && OptRight == KI::OptRight::Put)
120 ) {
121 S_max = Kokkos::fmax(S_max, S);
122 }
123 // Group all options that care about the path FLOOR
124 else if constexpr (
125 // Barrier types
126 OptType == KI::OptType::BarrierDownAndOut ||
127 OptType == KI::OptType::BarrierDownAndIn ||
128 // Lookback types
129 (OptType == KI::OptType::LookbackFixedStrike && OptRight == KI::OptRight::Put) ||
130 (OptType == KI::OptType::LookbackFloatingStrike && OptRight == KI::OptRight::Call)
131 ) {
132 S_min = Kokkos::fmin(S_min, S);
133 }
134 // Fallback, path-independent option not requiring tracking .
135 else {
136 // European,
137 // DigitalCashOrNothing,
138 // DigitalAssetOrNothing
139 }

142 // Final Payoff Resolver (To be called outside the time loop)
157 KOKKOS_INLINE_FUNCTION KT::Real evaluate_final_payoff(const KT::Real S,
158 const KT::Real Strike,
159 const KT::Real BarrierPrice,
160 const int n_t_steps) const {
161 KT::Real payoff = KT::real_zero;
162 // --------------------------------------------------------
163 // STANDARD OPTIONS LOGIC
164 // --------------------------------------------------------
165 if constexpr (OptType == KI::OptType::European) {
166 payoff = Payoff<OptRight>::evaluate_payoff(S, Strike);
167 } else if constexpr (OptType == KI::OptType::Asian) {
168 const double avg_price_db = S_sum / static_cast<double>(n_t_steps);
169 const auto avg_price = static_cast<KT::Real>(avg_price_db);
170 payoff = Payoff<OptRight>::evaluate_payoff(avg_price, Strike);
171 }
172 // --------------------------------------------------------
173 // BARRIER OPTIONS LOGIC
174 // --------------------------------------------------------
175 else if constexpr (OptType == KI::OptType::BarrierUpAndOut) {
176 if (S_max < BarrierPrice) {
177 // Survived
178 payoff = Payoff<OptRight>::evaluate_payoff(S, Strike);
179 }
180 } else if constexpr (OptType == KI::OptType::BarrierUpAndIn) {
181 if (S_max >= BarrierPrice) {
182 // Activated
183 payoff = Payoff<OptRight>::evaluate_payoff(S, Strike);
184 }
185 } else if constexpr (OptType == KI::OptType::BarrierDownAndOut) {
186 if (S_min > BarrierPrice) {
187 // Survived
188 payoff = Payoff<OptRight>::evaluate_payoff(S, Strike);
189 }
190 } else if constexpr (OptType == KI::OptType::BarrierDownAndIn) {
191 if (S_min <= BarrierPrice) {
192 // Activated
193 payoff = Payoff<OptRight>::evaluate_payoff(S, Strike);
194 }
195 }
196 // --------------------------------------------------------
197 // LOOKBACK OPTIONS LOGIC
198 // --------------------------------------------------------
199 else if constexpr (OptType == KI::OptType::LookbackFixedStrike) {
200 // Fixed Strike: Substitute S_max or S_min as the Reference Price
201 if constexpr (OptRight == KI::OptRight::Call) {
202 payoff = Payoff<OptRight>::evaluate_payoff(S_max, Strike);
203 } else {
204 // Put
205 payoff = Payoff<OptRight>::evaluate_payoff(S_min, Strike);
206 }
207 } else if constexpr (OptType == KI::OptType::LookbackFloatingStrike) {
208 // Floating Strike: Reference Price is S, Strike is substituted with S_min or S_max
209 if constexpr (OptRight == KI::OptRight::Call) {
210 // Evaluates: max(S - S_min, 0)
211 payoff = Payoff<OptRight>::evaluate_payoff(S, S_min);
212 } else {
213 // Put
214 // Evaluates: max(S_max - S, 0)
215 payoff = Payoff<OptRight>::evaluate_payoff(S, S_max);
216 }
217 }
218 // --------------------------------------------------------
219 // BINARY OPTIONS LOGIC
220 // --------------------------------------------------------
221 else if constexpr (OptType == KI::OptType::BinaryCashOrNothing) {
222 // Normalized to pay exactly 1.0 unit of cash if In-The-Money
223 constexpr KT::Real cash_payout = 1.0;
224 if constexpr (OptRight == KI::OptRight::Call) {
225 payoff = (S > Strike) ? cash_payout : KT::real_zero;
226 } else {
227 // Put
228 payoff = (S < Strike) ? cash_payout : KT::real_zero;
229 }
230 } else if constexpr (OptType == KI::OptType::BinaryAssetOrNothing) {
231 // Pays the actual terminal asset price if In-The-Money
232 if constexpr (OptRight == KI::OptRight::Call) {
233 payoff = (S > Strike) ? S : KT::real_zero;
234 } else {
235 // Put
236 payoff = (S < Strike) ? S : KT::real_zero;
237 }
238 }
239
240
241 return payoff;
242 }
243 };
