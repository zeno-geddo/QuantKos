It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

109 inline bool run_test() {
110 const std::string_view indent = " ";
111 bool all_tests_passed = true;
112 auto config = getDefaultConfigGoodIntegrand();
113
114 // Reference values from Rouah, pag 213
115 const std::map<double, double> S0_putPrice_map{
116 {8., 1.99958},
117 {9., 1.103571},
118 {10., 0.519039},
119 {11., 0.226137},
120 {12., 0.082123},
121 };
122
123 for (const auto pair: S0_putPrice_map) {
124 const double S0 = pair.first;
125 const double expected_price = pair.second;
126
127 config.market.S0 = S0;
128 std::cout << "\n\n" << indent << ">>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>\n";
129 std::cout << indent << "[ INFO ] S0 : " << S0 << " ; expected price : " << expected_price << " \n";
130
131 // Run Simulation
132 std::cout << indent << ">>> Calling the solver ...\n";
133 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
134 const auto MCRes = KTU::run_simulation(config);
135
136 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
137 std::cout << indent << ">>> Go back to the tester ...\n";
138
139 // Check absolute erroror
140 const KT::Real num_price = MCRes.OptionPrice.option_price;
141 const KT::Real abs_error = std::abs(num_price - expected_price);
142
143 // Statistical Tolerance (3-Sigma Rule: 99.7% Confidence Interval)
144 const KT::Real stat_error = MCRes.OptionPrice.standard_error;
145 const KT::Real stat_tolerance = 3.0 * stat_error;
146 // Discretization Bias Tolerance
147 const KT::Real bias_tolerance = std::max(expected_price * 0.1, 0.1);
148 // Total allowed tolerance
149 const KT::Real total_tolerance = stat_tolerance + bias_tolerance;
150
151
152 std::cout << indent << " Expe price: " << expected_price
153 << " | Num Price: " << num_price
154 << " | Error: " << abs_error
155 << " | Stat Error: " << stat_error
156 << " | Bias: " << bias_tolerance
157 << " | Allowed Tol Err: " << total_tolerance << "\n";
158
159 if (abs_error > total_tolerance) {
160 std::cerr << indent << "[ FAILED ] Error exceeds composite tolerance!" << std::endl;
161 all_tests_passed = false;
162 } else {
163 std::cout << indent << "[ PASSED ] Result is within statistical bounds." << std::endl;
164 }
165 }
166 return all_tests_passed;
167 }
