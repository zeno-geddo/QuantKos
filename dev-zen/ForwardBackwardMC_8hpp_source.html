QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

104 MCResults get_option_price() const {
105 // Initialize Helper Classes
106 // IMPORTANT: We force the forward solver to act like a European option (simply evaluate the forward paths).
107 BackwardLSMMemory Mem(config);
108 RNGManager RNGen(config);
109 MSolver<ModelPolicy, SchemePolicy, KI::OptType::European, OptRight> Solver(config);
110 OptionPricer OPricer(config);
111 KIO::OutputManager OWriter(config);
112 ForwardMCProgressTracker MCTracker(config, Mem.BatchMem);
113
114 // PHASE 1: Forward Batch Generation
115 MCTracker.print_pre_execution_diagnostic();
116 OWriter.print_paths_info_planned_outputs();
117 run_forward_phase(Mem, RNGen, Solver, OWriter, MCTracker);
118
119 // PHASE 2: Backward Induction (Longstaff-Schwartz regression)
120 run_backward_phase(Mem);
121
122 // 4. PHASE 3: Compute Final Option Price
123 // Feed the optimized backward cashflows into the standard pricer
124 OPricer.accumulate_batch_payoffs(Mem.h_best_future_outcomes, config.mc.N_Paths);
125 OPricer.evaluate_option_price();
126 OPricer.print_info_option_price();
127
128 return MCResults{config, OPricer.get_option_price_data()};
129 }

168 // ====================================================================
169 // PHASE 2: BACKWARD INDUCTION (Read from CPU RAM)
170 // ====================================================================
178 void run_backward_phase(BackwardLSMMemory &Mem) const {
179 const int nT = config.time.N_time_steps;
180 const KT::Real strike_price = config.options.StrikePrice;
181 const KT::Real discount_factor = Kokkos::exp(-config.market.r * config.time.dt);
182
183 using EngineLSM = LSM::LSMEngine<OptRight>;
184
185 std::cout << " >>> Starting Phase 2: Backward Induction (PCIe Streaming)...\n";
186 BackwardLSMProgressTracker BTracker(config);
187 BTracker.start_tracking();
188
189 // Initialization (t = T-1)
190 Mem.bring_host_prices_time_slice_to_device(nT - 1);
191 EngineLSM::initialize_cashflows(Mem.d_prices_current_time,
192 Mem.d_best_future_outcomes,
193 strike_price);
194 BTracker.update_progress();
195
196 // THE BACKWARD TIME MARCH
197 for (int t = nT - 2; t > 0; --t) {
198 // A. Memory Orchestration: Stream CPU RAM -> GPU VRAM
199 Mem.bring_host_prices_time_slice_to_device(t);
200
201 // B. Math Orchestration: Regression
202 const auto ls_coeffs = EngineLSM::perform_cross_paths_regression(Mem.d_prices_current_time,
203 Mem.d_best_future_outcomes,
204 discount_factor,
205 strike_price);
206
207 // C. Math Orchestration: Early Exercise Evaluation
208 EngineLSM::update_cashflows(Mem.d_prices_current_time,
209 Mem.d_best_future_outcomes,
210 ls_coeffs,
211 discount_factor,
212 strike_price);
213 BTracker.update_progress();
214 }
215
216 // 3. FINALIZATION (t = 0)
217 EngineLSM::apply_final_discount(Mem.d_best_future_outcomes, discount_factor);
218
219 // Pull final results back to CPU
220 Mem.bring_device_cash_flows_to_host();
221
222 // Stop tracking the backward phase (safe since no prints in between)
223 BTracker.finalize_tracking();
224 }
225 };
