It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

157 template<size_t NBasis>
158 struct LaguerreBasis {
171 KOKKOS_INLINE_FUNCTION
172 static void evaluate(const KT::Real X, KT::Real (&L)[NBasis]) {
173 // Older form
174 //L[0] = 1.0;
175 //if constexpr (NBasis > 1) L[1] = 1.0 - X;
176 //if constexpr (NBasis > 2) L[2] = 1.0 - 2.0 * X + 0.5 * X * X;
177 //if constexpr (NBasis > 3) L[3] = 1.0 - 3.0 * X + 1.5 * X * X - (X * X * X) / 6.0;
178 //if constexpr (NBasis > 4)
179 // L[4] = 1.0 - 4.0 * X + 3.0 * X * X - (2.0 / 3.0) * X * X * X + (1.0 / 24.0) * X *
180 // X * X * X;
181
182 // Homer's form
183 L[0] = KT::real_one;
184
185 if constexpr (NBasis > 1) L[1] = KT::real_one - X;
186
187 if constexpr (NBasis > 2)
188 L[2] = ((KT::real_05 * X - KT::real_two) * X) + KT::real_one;
189
190 if constexpr (NBasis > 3)
191 L[3] = (((static_cast<KT::Real>(-1.0 / 6.0) * X + KT::real_1p5) * X
192 - KT::real_3) * X) + KT::real_one;
193
194 if constexpr (NBasis > 4)
195 L[4] = ((((static_cast<KT::Real>(1.0 / 24.0) * X - static_cast<KT::Real>(2.0 / 3.0)) * X
196 + KT::real_3) * X - KT::real_4) * X)
197 + KT::real_one;
198 }
199 };

291 // 2. Get the coefs, C = M^-1 B
292 // Solve the system containing the (NBasisxNBasis) matrix on the CPU Host
293 return solve_system_on_host_cholesky(sums);

297 // =================================================================
298 // STEP 3: The Early Exercise Decision
299 // =================================================================
311 static void update_cashflows(const DevView1D &d_slice_prices_at_target_time,
312 const DevView1D &d_best_future_outcomes,
313 const Coeffs<NBasis> ls_coeffs,
314 // Pass by value since going to GPU and small struct
315 const KT::Real discount_factor,
316 const KT::Real strike_price) {
317 const int N = d_slice_prices_at_target_time.extent(0);
318
319 Kokkos::parallel_for("LSM_Update_Cashflows", N, KOKKOS_LAMBDA(const int i)
320 {
321 // Get variables needed for the exercise decision
322 const KT::Real S = d_slice_prices_at_target_time(i);
323 const KT::Real intrinsic_val = Payoff<OptRight>::evaluate_payoff(S, strike_price);
324 const KT::Real discounted_future_cf = d_best_future_outcomes(i) * discount_factor;
325
326 // Early exercise decision (ITM branch)
327 if (intrinsic_val > KT::real_zero) {
328 const KT::Real expected_val_of_holding = evaluate_expected_val_of_holding(ls_coeffs, S, strike_price);
329 // Continuation value
330 d_best_future_outcomes(i) = (intrinsic_val > expected_val_of_holding)
331 ? intrinsic_val
332 : discounted_future_cf;
333 // if : intrinsic_val > expected_val_of_holding the option is exercised (holding will be statistically worst). The future cash flow is overwritten with the intrinsic_val of today
334 // else : we hold the option, based on statistical average of all paths, holding will likely yield a bigger payout in the future
335 } else {
336 d_best_future_outcomes(i) = discounted_future_cf; // Hold since exiting is impossible
337 }
338 }
339 )
340 ;
341 // Kokkos::fence(); // // Fence not needed since there is a deep copty after
342 }

362 ;

378 static RegressionSums<NBasis> compute_regression_sums_on_device(const DevView1D &d_slice_prices,
379 const DevView1D &d_best_future_outcomes,
380 const KT::Real discount_factor,
381 const KT::Real strike_price) {
382 // Compute terms M and B of linear system MC = B
383 // NOTE : Method kept public since if private cuda does not have access to it
384
385 RegressionSums<NBasis> total_sums;
386 const int N_prices = d_slice_prices.extent(0);
387
388 Kokkos::parallel_reduce("LSM_Compute_Regression_Sums", N_prices,
389 KOKKOS_LAMBDA(const int i, RegressionSums<NBasis> & regr_sum)
390 {
391 const KT::Real S = d_slice_prices(i);
392 const KT::Real intrinsic_val = Payoff<OptRight>::evaluate_payoff(S, strike_price);
393
394 if (intrinsic_val > KT::real_zero) {
395 const KT::Real Y = d_best_future_outcomes(i) * discount_factor;
396 const KT::Real X = S / strike_price;
397
398 // Evaluate the basis polynomials into a local stack buffer
399 KT::Real L[NBasis];
400 LaguerreBasis<NBasis>::evaluate(X, L);
401
402 // Update M and B using the unified templated logic
403 // The compiler should unroll these loops, making it as fast as manual coding
404 for (size_t row = 0; row < NBasis; ++row) {
405 regr_sum.B(row) += static_cast<double>(L[row]) * Y;
406 for (size_t col = 0; col < NBasis; ++col) {
407 regr_sum.M(row, col) += static_cast<double>(L[row]) * static_cast<double>(L[col]);
408 }
409 }
410 regr_sum.count += 1.0;
411 }
412 }
413 ,
414 total_sums
415 )
416 ;
417 return total_sums;
418 }

429 KOKKOS_INLINE_FUNCTION
430 static double evaluate_expected_val_of_holding(const Coeffs<NBasis> &coeffs,
431 const KT::Real S,
432 const KT::Real strike_price) {
433 const KT::Real X = S / strike_price;
434 KT::Real L[NBasis];
435
436 // Use the unified basis evaluator
437 LaguerreBasis<NBasis>::evaluate(X, L);
438
439 double continuation_value = 0.0;
440
441 // Accumulate the weighted sum (coeffs * basis)
442 for (size_t i = 0; i < NBasis; ++i) {
443 continuation_value += coeffs.C[i] * static_cast<double>(L[i]);
444 }
445
446 return continuation_value;
447 }

460 static Coeffs<NBasis> solve_system_on_host_cholesky(const RegressionSums<NBasis> &sums) {
461 // If not enough paths to form a basis, return zero coefficients
462 if (sums.count < static_cast<double>(NBasis)) return Coeffs<NBasis>{};
463
464 // Tikhonov Regularization
465 // Add a tiny amount of noise (1 part per million) relative to the signal present in the data (average diagonal element)
466 // KT::Real trace = 0.0;
467 // for (size_t i = 0; i < NBasis; ++i) {
468 // trace += sums.M(i, i);
469 // }
470 // const KT::Real avg_diag = trace / static_cast<KT::Real>(NBasis);
471 // constexpr KT::Real eps_factor = std::is_same_v<KT::Real, float> ? 1e-4 : 1e-6;
472 // const KT::Real reg = std::max(eps_factor * avg_diag, eps_factor);
473
474
475
476 // 1. Prepare local M and B
477 constexpr double reg = 1e-8;
478 double M[NBasis][NBasis];
479 double B[NBasis];
480
481 for (size_t i = 0; i < NBasis; ++i) {
482 B[i] = sums.B(i);
483 for (size_t j = 0; j < NBasis; ++j) {
484 M[i][j] = sums.M(i, j) + ((i == j) ? reg : 0.0);
485 }
486 }
487
488 // 2. Cholesky Decomposition: M = L * L^T
489 double L[NBasis][NBasis] = {0.0}; // Here L stands for lower
490
491 constexpr double singularity_limit = 1e-14;
492
493 for (size_t i = 0; i < NBasis; ++i) {
494 for (size_t j = 0; j <= i; ++j) {
495 double s = 0.0;
496 for (size_t k = 0; k < j; ++k) s += L[i][k] * L[j][k];
497
498 if (i == j) {
499 const double val = M[i][i] - s;
500 if (val < singularity_limit) return Coeffs<NBasis>{}; // Not pos-def
501 L[i][i] = std::sqrt(val);
502 } else {
503 L[i][j] = (M[i][j] - s) / L[j][j];
504 }
505 }
506 }
507
508 // 3. Forward substitution: L * y = B
509 double y[NBasis];
510 for (size_t i = 0; i < NBasis; ++i) {
511 double s = 0.0;
512 for (size_t k = 0; k < i; ++k) s += L[i][k] * y[k];
513 y[i] = (B[i] - s) / L[i][i];
514 }
515
516 // 4. Backward substitution: L^T * C = y
517 Coeffs<NBasis> C;
518 for (int i = static_cast<int>(NBasis) - 1; i >= 0; --i) {
519 double s = 0.0;
520 for (size_t k = i + 1; k < NBasis; ++k) s += L[k][i] * C[k];
521 C[i] = (y[i] - s) / L[i][i];
522 }
523
524 return C;
525 }
526 };
