It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

139 void save_paths_batch_if_needed(const int current_batch_size,
140 const Engine::PathsMCBatchMem &BatchMem) override {
141 // Return without writing if no file is specified
142 if (config.output.filename_paths_out.empty()) {
143 return;
144 }
145
146 //Calculate exactly how many bytes this specific batch occupies (N_sims x n_t_steps x sizeReal)
147 // This is flexible and allows to consider cases where the batch is not complete
148 const size_t bytes_to_write = static_cast<size_t>(current_batch_size) *
149 static_cast<size_t>(config.time.N_time_steps) * sizeof(KT::Real);
150
151 // Fetch the raw C-style pointer from the Kokkos Host View
152 const KT::Real *raw_data_ptr = BatchMem.h_batch_view.data();
153
154 // SCENARIO A: The Fast Track (Only runs if the layout is physically contiguous and layout right)
155 if (BatchMem.is_host_contiguous() && BatchMem.is_host_row_major()) {
156 // Dump it to the SSD in one hardware instruction
157 out_paths_stream.write(reinterpret_cast<const char *>(raw_data_ptr), bytes_to_write);
158 }
159
160 // SCENARIO B: The Universal Buffered Fallback
161 else {
162 // 1. Allocate a contiguous memory buffer for exactly one row
163 std::vector<KT::Real> row_buffer(config.time.N_time_steps);
164 const size_t row_bytes = config.time.N_time_steps * sizeof(KT::Real);
165
166 // Loop through row-by-row
167 for (int i = 0; i < current_batch_size; ++i) {
168 // 2. Read the scattered RAM data (whatever layout it is) into our clean buffer
169 for (int j = 0; j < config.time.N_time_steps; ++j) {
170 row_buffer[j] = BatchMem.h_batch_view(i, j);
171 }
172
173 // 3. Dump the ENTIRE ROW to the SSD in exactly one hardware instruction
174 out_paths_stream.write(reinterpret_cast<const char *>(row_buffer.data()), row_bytes);
175 }
176 }
177 }

239 BinHeader header;

297 std::vector<KT::Real> read_prices_at_target_time(const KT::Real current_time) override {
298 if (!inp_stream.is_open()) {
299 throw std::runtime_error("[BinReader Error] Cannot read prices, file stream is closed.");
300 }
301
302 // Map continuous time to discrete index (a row major layour is assumed)
303 const int target_col_idx = static_cast<int>(std::round(current_time / bin_header.dt)) - 1;
304 // Check That required time is in time domain
305 if (target_col_idx < 0 || target_col_idx >= bin_header.n_time_steps) {
306 throw std::runtime_error("[BinReader Error] Requested time " + std::to_string(current_time) +
307 " is outside the simulated time grid bounds.");
308 }
309 // Check that the required time is in the time grid and do not required interpolation
310 const double actual_grid_time = (target_col_idx + 1) * bin_header.dt;
311 if (std::abs(current_time - actual_grid_time) > 1e-10) {
312 throw std::runtime_error("Requested time " + std::to_string(current_time) +
313 " does not align perfectly with the discrete simulation grid (dt = " +
314 std::to_string(bin_header.dt) + ")."
315 );
316 }
317
318 // Allocate the vector containing the prices at target times
319 std::vector<KT::Real> target_prices(bin_header.total_n_sims);
320
321 // Calculate the memory strides
322 const std::streampos start_offset_bytes = sizeof(BinHeader) + (target_col_idx * sizeof(KT::Real));
323 //Absolute location in a file from beginning
324 const std::streamoff row_stride_bytes = bin_header.n_time_steps * sizeof(KT::Real);
325 // jumping of a simulation (relative distance)
326
327 // Read Prices
328 const double scale = bin_header.scale_factor;
329 for (int i = 0; i < bin_header.total_n_sims; ++i) {
330 // Calculate absolute byte position for this specific element by adding a relative distance (streamoff) to the starting point
331 std::streampos element_pos = start_offset_bytes + (static_cast<std::streamoff>(i) * row_stride_bytes);
332 // Jump to the exact byte location on the disk (Start measuring the element position from beginning of the file)
333 inp_stream.seekg(element_pos, std::ios::beg);
334 // Read exactly one floating point number into our vector
335 inp_stream.read(reinterpret_cast<char *>(&target_prices[i]), sizeof(KT::Real));
336 // Multiply in-place to scale the price
337 target_prices[i] = static_cast<KT::Real>(static_cast<double>(target_prices[i]) * scale);
338 }
339
340 // Clear any EOF flags that might have triggered, allowing future reads
341 inp_stream.clear();
342
343 return target_prices;
344 }

356 void load_header_paths_file() {
357 // Open File Paths
358 auto file_out_paths = std::filesystem::path(config.output.out_dir) / config.output.filename_paths_out;
359 inp_stream.open(file_out_paths.string(), std::ios::in | std::ios::binary);
360 if (!inp_stream.is_open()) {
361 throw std::runtime_error("[BinReader Error] Failed to open file: " + file_out_paths.string());
362 }
363 // Read header
364 inp_stream.read(reinterpret_cast<char *>(&bin_header), sizeof(BinHeader));
365 // Check if header is ok
366 if (bin_header.file_key[0] != 'Q' || bin_header.file_key[1] != 'U' ||
367 bin_header.file_key[2] != 'A' || bin_header.file_key[3] != 'N' ||
368 bin_header.file_key[4] != 'T' || bin_header.file_key[5] != 'K' ||
369 bin_header.file_key[6] != 'O' || bin_header.file_key[7] != 'S') {
370 throw std::runtime_error(
371 "[BinReader Error] Magic key mismatch. File is corrupted or not a QuantKos.paths file.");
372 }
373 if (bin_header.byte_precision != sizeof(KT::Real)) {
374 throw std::runtime_error(
375 "[BinReader Error] Build precision mismatch. Data cannot safely map into target system precision.");
376 }
377 }
