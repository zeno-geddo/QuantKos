QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

82 int run_tests(const int argc, char *argv[]) {
83 // Check for help flag
84 if (handle_help(argc, argv)) {
85 return 0;
86 }
87
88 // Determine tests to run
89 std::vector<std::string> id_tests_to_run;
90 if (argc > 1) {
91 try {
92 // Try to see in the yalm configuration file which are the test to run
93 std::cout << "Opening YAML file at: " << argv[1] << std::endl;
94 const YAML::Node tests_config = YAML::LoadFile(argv[1]);
95 for (const auto &node: tests_config["run_tests"]) {
96 id_tests_to_run.push_back(node.as<std::string>());
97 }
98 } catch (...) {
99 std::cerr << "Error: Could not parse YAML file or file not found.\n";
100 std::cerr << get_help_message() << std::endl;
101 return 1;
102 }
103 } else {
104 // Run all tests if not yalm file is passed
105 for (const auto &[id_test, test_suite]: tests_registry) {
106 id_tests_to_run.push_back(id_test);
107 }
108 }
109
110 // Execute tests
111 int n_tests_failed = 0;
112 for (const auto &id_test: id_tests_to_run) {
113 const bool test_required = static_cast<bool>(tests_registry.count(id_test));
114 // 1 if there is the key, 0 else
115 if (test_required) {
116 const auto &target_test_suite = tests_registry.at(id_test);
117 try {
118 const bool test_passed = target_test_suite.test_function();
119 if (test_passed) {
120 std::cout << indent << "[ SUCCESS ] " << target_test_suite.description <<
121 " Suite passed successfully.\n\n";
122 } else {
123 std::cout << indent << "[ TEST FAILURE ] " << target_test_suite.description <<
124 " Suite detected an error!\n\n";
125 std::cerr << indent << "[ TEST FAILURE ] " << target_test_suite.description <<
126 " Suite detected an error!\n\n";
127 n_tests_failed++;
128 }
129 } catch (const std::runtime_error &e) {
130 std::cout << indent << "[ FATAL RUNTIME ERROR ] " << target_test_suite.description
131 << " Suite made the software crash failed with: \n\n " << e.what() << "\n\n";
132 std::cerr << indent << "[ FATAL RUNTIME ERROR ] " << target_test_suite.description
133 << " Suite made the software crash failed with: \n\n " << e.what() << "\n\n";
134 n_tests_failed++;
135 } catch (...) {
136 std::cout << indent << "[ FATAL ERROR ] " << target_test_suite.description <<
137 " Suite made the software crash!\n\n";
138 std::cerr << indent << "[ FATAL ERROR ] " << target_test_suite.description <<
139 " Suite made the software crash!\n\n";
140 n_tests_failed++;
141 }
142 }
143 }
144 return n_tests_failed;
145 }

149 // Start with a short explanation
150 std::ostringstream oss;
151 oss << "Usage: ./Test [YAML_CONFIG_FILE] [--help]\n\n"
152 "Options:\n" <<
153 indent << "--help Display this help message.\n\n"

157 " - \"test_id_1\"\n"
158 " - \"test_id_2\"\n\n"
159 "If no YAML file is provided, the engine will execute all registered tests.\n"
160 "List Available tests :\n";
161 // Show the id of the implemented tests
162 for (const auto& [id, suite] : tests_registry) {
163 oss << indent << " - " << id << " # " << suite.description + "\n";
164 }
165
166 return oss.str();
167 }
