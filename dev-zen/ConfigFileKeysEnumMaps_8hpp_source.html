QuantKos v0.1.0 (Beta)
QuantKos is a high-performance Monte Carlo option pricing engine written in modern C++20 and built on top of the Kokkos performance portability framework. It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

124 // --------------------------------------------------------
125 // 2. Enum -> String
126 // --------------------------------------------------------
133 template<typename T>
134 std::string enum_to_string(T value) {
135 const auto &m = StrEnumMap<T>::get();
136 for (const auto &pair: m) {
137 if (pair.second == value) {
138 return pair.first;
139 }
140 }
141 return "Unknown";
142 }

164 template<typename T>
165 T string_to_enum(const std::string &input, const std::string &field_name) {
166 const auto &m = StrEnumMap<T>::get();
167 auto it = m.find(input);
168
169 if (it != m.end()) {
170 return it->second;
171 }
172
173 // Error handling inside the utility
174 std::string err = "Config Error: Invalid value '" + input + "' for " + field_name + ".\n";
175 err += "Allowed options: [ ";
176 for (const auto &pair: m) err += pair.first + " ";
177 err += "]";
178 throw std::runtime_error(err);
179 }
