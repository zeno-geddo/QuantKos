136 inline bool run_test_gaussian() {
137 std::string_view indent{" "};
138 std::cout << "\n\n" << indent << "====================================================================\n"
139 << indent << " TEST 1 : Standard Normal Distribution Check \n"
140 << indent << "====================================================================\n";
141
142 std::cout << indent << "[ RUN ] Standard Normal Distribution Check\n";
143
144
145 bool test_passed = true;
146
147 // Define size test
148 int batch_size = 1'000'000;
149 int N_time_steps = 365;
150
151 // Initialize Random number generation manager
152 const auto rng_man = initialize_rng_manager();
153
154 // Allocate Memory for the generated random number
155 auto d_z_values = Kokkos::View<KT::Real **>("device_Z_values",
156 batch_size,
157 N_time_steps);
158 auto h_z_values = Kokkos::create_mirror_view(d_z_values);
159
160 // Launch Dummy Kernel to see if we are really generating guassina values
161 execute_rng_kernel(batch_size,
162 N_time_steps,
163 d_z_values,
164 rng_man.get_global_rng_pool());
165 // Copy results on host if not already there
166 Kokkos::deep_copy(h_z_values, d_z_values);
167
168 // Calculate Statistics
169 double sum = 0.0;
170 double sq_sum = 0.0;
171
172 for (int i = 0; i < batch_size; ++i) {
173 for (int j = 0; j < N_time_steps; ++j) {
174 double z = static_cast<double>(h_z_values(i, j));
175 sum += z;
176 sq_sum += (z * z);
177 }
178 }
179
180 double N_draws = static_cast<double>(batch_size * N_time_steps);
181 double mean = sum / N_draws;
182 double variance = (sq_sum / N_draws) - (mean * mean);
183
184 std::cout << indent << "[ INFO ] Number Samples : " << N_draws << "\n";
185 std::cout << indent << "[ INFO ] Measured Mean : " << mean << "\n";
186 std::cout << indent << "[ INFO ] Measured Variance : " << variance << "\n";
187
188 if (std::abs(mean - 0.0) > 0.005) {
189 std::cerr << indent << "[ FAILED ] RNG Mean drifted unacceptably far from 0.0!\n";
190 test_passed = false;
191 }
192
193 if (std::abs(variance - 1.0) > 0.01) {
194 std::cerr << indent << "[ FAILED ] RNG Variance drifted unacceptably far from 1.0!\n";
195 test_passed = false;
196 }
197
198 if (test_passed) {
199 std::cout << indent << "[ PASSED ] Standard Normal Distribution Check \n";
200 }
201
202
203 std::cout << indent << "====================================================================\n" << std::endl;
204
205 return test_passed;
206 }
