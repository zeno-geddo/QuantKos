63 inline std::vector<int> get_time_grid_resolutions(const int lower_exponent = 3, const int upper_exponent = 9) {
64 // Return a vector containing different resolution (Calculated as 2^i) for the grid
65 if (lower_exponent > upper_exponent)
66 throw std::runtime_error("Error : Lower exponent is greater than upper exponent");
67
68 std::vector<int> steps;
69 steps.reserve(upper_exponent - lower_exponent + 1);
70
71 for (int i = lower_exponent; i <= upper_exponent; ++i) {
72 steps.push_back(1 << i);
73 }
74
75 return steps;
76 }

109 [[nodiscard]] inline OptionPriceErr compare_numerical_and_expected_option_price(quantkos::Config::UInputs &config,
110 const double expected_price,
111 const std::string_view indent = " ") {
112 std::cout << indent << ">>> Calling the solver ...\n";
113 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
114
115 const auto MCRes = run_simulation(config);
116 const double num_price = MCRes.OptionPrice.option_price;
117 const double stat_error = MCRes.OptionPrice.standard_error;
118 const double signed_error = num_price - expected_price;
119 const double abs_error = std::abs(signed_error);
120 const double z_score = signed_error / stat_error;
121
122 std::cout << indent << "Resolution considering " << config.time.N_time_steps << " times steps:\n";
123 std::cout << indent << " Price : " << num_price << '\n';
124 std::cout << indent << " Reference : " << expected_price << '\n';
125 std::cout << indent << " Error : " << signed_error << '\n';
126 std::cout << indent << " Abs Error : " << abs_error << '\n';
127 std::cout << indent << " Std Error : " << stat_error << '\n';
128 std::cout << indent << " Z-score : " << z_score << '\n';
129
130 if (abs_error <= 3.0 * stat_error) {
131 std::cout << indent << "[ PASSED ] Numerical results statistically consistent with reference.\n";
132 } else {
133 std::cout << indent << "[ INFO ] Bias still dominates. Continue refining dt. \n";
134 }
135
136 // Return the results of the test and go back to tester for next text
137 std::cout << "\n" << indent << "--------------------------------------------------------------------\n";
138 std::cout << indent << ">>> Go back to the tester ...\n";
139
140 return {config.time.N_time_steps, config.time.inp_dt, signed_error, stat_error};
141 }

143 // Analyzes convergence data, prints diagnostics, and returns true if all Z-scores are within 3-sigma bounds.
158 [[nodiscard]] inline bool evaluate_quality_numerical_results(const std::vector<OptionPriceErr> &convergence_results,
159 const std::string_view indent = " ") {
160 if (convergence_results.empty()) return false;
161
162 bool test_passed = true;
163
164 std::cout << indent << "\n::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::\n";
165 std::cout << indent << "Analyzing convergence order for given model and scheme...\n";
166 std::cout << indent <<
167 "Total Error = Discretization Error (O(dt)) + Statistical Noise (O(std/sqrt(N_paths)))\n\n";
168
169 for (size_t i = 0; i < convergence_results.size(); ++i) {
170 const auto &current = convergence_results[i];
171 const double z_score = current.error / current.stat_error;
172
173 std::cout << indent << "[ INFO ] Step " << i + 1
174 << ": Ndt= " << current.N_dt
175 << ", ERR= " << current.error
176 << ", STAT ERR= " << current.stat_error
177 << ", Zscore= " << z_score << "\n";
178
179 // Compute convergence order only if there is a subsequent step to compare against
180 if (i < convergence_results.size() - 1) {
181
182 const auto &next = convergence_results[i + 1];
183 const double log_dt_diff = std::log(next.dt) - std::log(current.dt);
184 const double log_err_diff = std::log(std::abs(next.error)) - std::log(std::abs(current.error));
185 const double order = log_err_diff / log_dt_diff;
186
187 // Show convergence and warn if the error hit the statistical noise floor
188 if (std::abs(current.error) > current.stat_error) {
189 std::cout << indent << "[ INFO ] Step " << i + 1 << " -> " << i + 2
190 << " Convergence Order: " << order << "\n";
191 } else {
192 std::cout << indent << "[ INFO ] Step " << i + 1 << " -> " << i + 2
193 << " Convergence Order: " << order << " (Warning : Statistical error dominate) \n";
194 }
195 } else { // Enter this block only if the last iteration (max resolution) is considered
196 // If the error exceeds 3 standard deviations from max resultion, the test mathematically fails.
197 if (std::abs(z_score) > 3.0) test_passed = false;
198 }
199
200 }
201
202 std::cout << indent << "::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::\n";
203
204 return test_passed;
205 }

221 [[nodiscard]] inline bool run_weak_convergence_test(
222 const std::string_view test_name,
223 KC::UInputs config,
224 const double expected_option_price,
225 const std::vector<std::pair<KI::MathModel, KI::NumScheme> > &models_to_test,
226 const std::vector<int> &time_grid_resolutions,
227 const std::string_view indent = " ") {
228 std::cout << "\n\n" << indent << "====================================================================\n"
229 << indent << " " << test_name << "\n"
230 << indent << "====================================================================\n";
231
232 std::cout << indent << "[ INFO ] Target Exact Price : " << expected_option_price << "\n";
233
234 bool all_tests_passed = true;
235
236 // Loop over models
237 for (const auto &model: models_to_test) {
238 config.model.id_model = model.first;
239 config.scheme.id_scheme = model.second;
240
241 std::vector<OptionPriceErr> convergence_results;
242
243 // Loop over time steps
244 for (int N_Tsteps: time_grid_resolutions) {
245 config.time.N_time_steps = N_Tsteps;
246 config.time.inp_dt = config.time.t_end / N_Tsteps;
247
248 std::cout << indent << "[ INFO ] MODEL : " << KI::enum_to_string(config.model.id_model) << "\n";
249 std::cout << indent << "[ INFO ] SCHEME : " << KI::enum_to_string(config.scheme.id_scheme) << "\n";
250 std::cout << indent << "[ INFO ] dt: " << std::fixed << std::setprecision(6) << config.time.inp_dt
251 << "\n";
252
253 // Run simulation
254 auto convergence_res = compare_numerical_and_expected_option_price(config, expected_option_price);
255 convergence_results.push_back(convergence_res);
256 }
257
258 // Check If numerical results are correct for this specific model/scheme combo
259 const bool model_passed = evaluate_quality_numerical_results(convergence_results);
260 if (!model_passed) {
261 all_tests_passed = false; // Mark the whole suite as failed if one combo fails
262 }
263 }
264
265 // Print final message and exit function
266 std::cout << "\n" << indent << "====================================================================\n";
267 if (all_tests_passed) {
268 std::cout << indent << "[ PASSED ] " << test_name << " successfully converged.\n";
269 } else {
270 std::cerr << indent << "[ FAILED ] " << test_name << " drifted outside statistical bounds.\n";
271 }
272 std::cout << indent << "====================================================================\n\n";
273
274 return all_tests_passed;
275 }
