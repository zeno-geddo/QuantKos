It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

49 inline bool run_test() {
50 std::string_view indent{" "};
51 std::cout << "\n\n" << indent << "====================================================================\n"
52 << indent << " TEST 2 : Binary I/O Round-Trip Check \n"
53 << indent << "====================================================================\n";
54 std::cout << indent << "[ RUN ] Binary I/O Round-Trip Check\n";
55
56 bool overall_test_passed = true;
57
58 // The target time we want to extract (e.g., t = 0.7 -> Index 6)
59 const double target_time = 0.7;
60 const int expected_col_idx = 6;
61
62 const std::vector<bool> norm_flags = {false, true};
63 for (bool f: norm_flags) {
64
65 // Show the normalization flag
66 bool current_run_passed = true;
67 std::cout << indent << "\n[ INFO ] Norm flag : " << f << "\n";
68
69 // 1. Setup a Predictable Configuration
70 // !!! NOTE : if this in put outside the loop, and the bool vect can stop commuting and the test can break !
71 auto config = KC::UInputs();
72 config.output.filename_paths_out = "IO_BIN_test_roundtrip.paths";
73 config.output.out_dir = "."; // Current directory
74 config.mc.N_Paths = 100;
75 config.mc.batch_size = 100;
76 config.time.N_time_steps = 10;
77 config.time.t_end = 1.0;
78 config.time.dt = 0.1;
79 config.mc.normalize_prices = f;
80 config.apply_price_scaling();
81
82
83 // ====================================================================
84 // PHASE 1: GENERATE AND WRITE FAKE DATA
85 // ====================================================================
86 std::cout << indent << "[ INFO ] Phase 1: Generating and Writing predictable matrix...\n";
87 {
88 // Allocate Memory Manager
89 KE::PathsMCBatchMem dummy_batch(config);
90
91 // Fill the host view with predictable data: value = (path_id * 1000) + time_step
92 for (int i = 0; i < config.mc.N_Paths; ++i) {
93 for (int j = 0; j < config.time.N_time_steps; ++j) {
94 const auto raw_val = static_cast<double>((i * 1000) + j);
95 const double scale_fact = config.get_price_scaling_factor();
96 dummy_batch.h_batch_view(i, j) = static_cast<KT::Real>(raw_val/scale_fact);
97 }
98 }
99
100 // Write to disk using BinWriter
101 KB::BinWriter writer(config);
102 writer.save_paths_batch_if_needed(config.mc.batch_size, dummy_batch);
103
104 // The dummy_batch is destroyed, and writer.~BinWriter() is called,
105 // safely flushing the bytes to the SSD and closing the file lock!
106 }
107
108
109 // ====================================================================
110 // PHASE 2: READ AND VALIDATE DATA
111 // ====================================================================
112 std::cout << indent << "[ INFO ] Phase 2: Reading back target time t = " << target_time << "...\n";
113
114 try {
115 KB::BinReader reader(config);
116 std::vector<KT::Real> extracted_prices = reader.read_prices_at_target_time(target_time);
117
118 // Assert size matches
119 if (extracted_prices.size() != static_cast<size_t>(config.mc.N_Paths)) {
120 std::cerr << indent << "[ FAILED ] Extracted vector size (" << extracted_prices.size()
121 << ") does not match expected (" << config.mc.N_Paths << ").\n";
122 current_run_passed = false;
123 }
124
125 // Assert data matches our predictable pattern exactly
126 for (int i = 0; i < config.mc.N_Paths; ++i) {
127 const auto expected_value = static_cast<KT::Real>((i * 1000) + expected_col_idx);
128
129 if (std::abs(extracted_prices[i] - expected_value) > 1e-3) { // Account for single precision induced errors
130 std::cerr << indent << "[ FAILED ] Data mismatch at path " << i
131 << ". Expected: " << expected_value << ", Got: " << extracted_prices[i] << "\n";
132 current_run_passed = false;
133 break;
134 }
135 }
136 } catch (const std::exception &e) {
137 std::cerr << indent << "[ FAILED ] Exception thrown during reading: " << e.what() << "\n";
138 current_run_passed = false;
139 }
140
141 if (current_run_passed) {
142 std::cout << indent << "[ OK ] Binary Writer/Reader round-trip succeeded for norm_flag = "
143 << (f ? "true" : "false") << ".\n";
144 } else {
145 overall_test_passed = false;
146 }
147
148 // ====================================================================
149 // PHASE 3: CLEANUP
150 // ====================================================================
151 std::filesystem::path file_out_paths =
152 std::filesystem::path(config.output.out_dir) / config.output.filename_paths_out;
153 if (std::filesystem::exists(file_out_paths)) {
154 std::filesystem::remove(file_out_paths);
155 }
156 }
157 std::cout << indent << "======================== FINISHED BINARY IO TEST========================================\n" <<
158 std::endl;
159 return overall_test_passed;
160 }
