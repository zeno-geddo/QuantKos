It provides a single-source implementation capable of targeting serial execution, multi-core CPUs, and modern GPUs while maintaining the same code base.

281 void print_progress_bar(const int n_current_batch) {
282 constexpr int bar_width = 20;
283 const int n_total_batches = batch_mem.total_batch_loops();
284
285 const float progress = static_cast<float>(n_current_batch) / static_cast<float>(n_total_batches);
286 const int bar_front = static_cast<int>(static_cast<float>(bar_width) * progress);
287 const int percentage_done = static_cast<int>(progress * 100.0f);
288
289 std::cout << " [";
290 for (int i = 0; i < bar_width; ++i) {
291 if (i < bar_front) std::cout << "#";
292 else if (i == bar_front) std::cout << ">";
293 else std::cout << ".";
294 }
295
296 std::cout << "] " << percentage_done << "% "
297 << "(" << n_current_batch << "/" << n_total_batches << " Batches)";
298
299 if (n_current_batch > 0) {
300 const auto current_time = std::chrono::steady_clock::now();
301 const std::chrono::duration<double> elapsed = current_time - start_time;
302 const double elapsed_seconds = elapsed.count();
303
304 const double total_estimated_time = elapsed_seconds / progress;
305 const double eta_seconds = total_estimated_time - elapsed_seconds;
306
307 std::cout << std::fixed << std::setprecision(1)
308 << "(T: " << elapsed_seconds << "s | "
309 << "ETA: " << (n_current_batch == n_total_batches ? 0.0 : eta_seconds) << "s)\r";
310 }
311 std::cout << "\r" << std::flush;
312 }

366 void update_progress() {
367 steps_completed++;
368
369 // Calculate current integer percentage
370 if (total_lsm_steps <= 0) return; // Prevent division by zero if there are almost no time steps
371 const int current_percent = static_cast<int>(
372 (static_cast<float>(steps_completed) / static_cast<float>(total_lsm_steps)) * 100.0f
373 );
374
375 // Print the progress bar if target percentage reached
376 if (current_percent >= last_printed_percent + update_interval_percent) {
377 print_progress_bar(steps_completed);
378 last_printed_percent = current_percent; // Update the threshold
379 }
380 }
381
385 void finalize_tracking() {
386 print_progress_bar(total_lsm_steps);
387 std::cout << "\n\n ========================================================\n" << std::endl;
388 }

392 int total_lsm_steps; //< Internal counter keeping the loop size (// T-1 regressions steps)

395 //< Frequency, in percentage of the total, when to print time steps
396 int last_printed_percent = 0;

489 template<typename SolverType, typename HostDeepCopy>
490 inline void run_forward_all_mc_batches(
491 PathsMCBatchMem &BatchMem,
492 const RNGManager &RNGen,
493 const SolverType &Solver,
494 KIO::OutputManager &OWriter,
495 ForwardMCProgressTracker &MCTracker,
496 HostDeepCopy &&custom_host_deep_copy // Lambda function to copy desired data to host
497 ) {
498 const int full_batch_size = BatchMem.n_sims_per_batch;
499 const int n_full_batches = BatchMem.n_full_batches();
500 const int n_sims_left_over = BatchMem.n_sims_left_over_after_full_batches();
501 const int n_total_batch_loops = BatchMem.total_batch_loops();
502
503 MCTracker.start_tracking();
504 for (int n_curr_batch = 0; n_curr_batch < n_total_batch_loops; ++n_curr_batch) {
505 const int current_batch_size = (n_curr_batch < n_full_batches) ? full_batch_size : n_sims_left_over;
506
507 // 1. Fire parallel integration kernels on the active Kokkos execution space (Device/GPU)
508 Solver.execute_batch(current_batch_size, BatchMem, RNGen);
509
510 // 2. Execute the custom host-device synchronization policy (e.g. DMA transfers, accumulation)
511 custom_host_deep_copy(n_curr_batch, current_batch_size); // Note : Injected via Lambda
512
513 // 3. Write path diagnostics/trajectories to the file system if configured
514 OWriter.save_paths_batch_if_needed(current_batch_size, BatchMem); //Save batch to disk
515
516 // 4. Advance progress and compute linear clock ETA using the tracker defined in the Canvas
517 MCTracker.update_progress(n_curr_batch + 1);
518 }
519 MCTracker.finalize_tracking();
520 }

560 template<typename MemoryType, typename PricingCallable>
561 inline MCResults execute_mc_framework(const KC::UInputs &config, PricingCallable &&pricing_function) {
562 // 1. Allocate Heavy Memory ONCE for the exact type requested
563 MemoryType Mem(config);
564
565 // Initialize container to track timing
566 MCTimer Timer;
567 Timer.start_total();
568
569 // 2. Get the option Price
570 Timer.start_pricing();
571 OptionPricer::MCOpPrices ref_option_price_data = pricing_function(config, Mem);
572 Timer.stop_pricing();
573
574 // 3. Compute Greeks if needed (Using the generic pricing callable)
575 MCGreeks greeks{};
576 if (config.must_compute_greeks()) {
577 // Track greeks timing
578 Timer.start_greeks();
579
580 // Create a copy where spoofing params common to all greeks
581 KC::UInputs base_greek_cfg = config;
582 base_greek_cfg.mc.analyze_risk_neutral_payoff_distribution = false; // payoffs never stored and sorted
583 base_greek_cfg.output.filename_paths_out = ""; // So that paths for the greeks are never saved
584 // should introduce and rewrite a flag that tell if to print or not, and during the greeks should be low verbosity
585
586 if (base_greek_cfg.mc.compute_delta_et_gamma) {
587 std::cout << "\n\n >>> Computing Delta & Gamma...\n";
588 // Get initial price and price bump considering that it could have been normalized
589 const auto S0 = base_greek_cfg.market.S0;
590 const auto h_S = S0 * base_greek_cfg.mc.spot_price_relative_bump_size;
591 auto h_S_real_scale = h_S;
592 if (base_greek_cfg.mc.normalize_prices) {
593 const auto S0_real_scale = base_greek_cfg.get_price_scaling_factor();
594 h_S_real_scale = S0_real_scale * base_greek_cfg.mc.spot_price_relative_bump_size;
595 }
596
597 KC::UInputs cfg_up = base_greek_cfg;
598 cfg_up.market.S0 = S0 + h_S;
599 double p_up = pricing_function(cfg_up, Mem).option_price;
600
601 KC::UInputs cfg_down = base_greek_cfg;
602 cfg_down.market.S0 = S0 - h_S;
603 double p_down = pricing_function(cfg_down, Mem).option_price;
604
605 // NOTE : Divide by the bump in the real scale to match the scale of the real output prices
606 greeks.delta = (p_up - p_down) / (2.0 * h_S_real_scale);
607 greeks.gamma = (p_up - 2.0 * ref_option_price_data.option_price + p_down) /
608 (h_S_real_scale * h_S_real_scale);
609 }
610
611 if (base_greek_cfg.mc.compute_vega_et_vomma) {
612 // Approach 1
613 std::cout << "\n\n >>> Computing Vega...\n";
614 const double var0 = base_greek_cfg.market.v0;
615 const double vol0 = std::sqrt(var0);
616 const double h_vol = base_greek_cfg.mc.volatility_absolute_bump_size;
617
618 KC::UInputs cfg_up = base_greek_cfg;
619 const double vol_up = vol0 + h_vol;
620 const double var_up = vol_up * vol_up;
621 cfg_up.market.v0 = var_up; // variance instead of volatility required
622 double p_up = pricing_function(cfg_up, Mem).option_price;
623
624 KC::UInputs cfg_down = base_greek_cfg;
625 const double vol_down = vol0 - h_vol;
626 const double var_down = vol_down * vol_down;
627 cfg_down.market.v0 = var_down; // variance instead of volatility required
628 double p_down = pricing_function(cfg_down, Mem).option_price;
629
630 greeks.vega = (p_up - p_down) / (2.0 * h_vol);
631 greeks.vomma = (p_up - 2.0 * ref_option_price_data.option_price + p_down) / (h_vol * h_vol);
632 }
633
634 if (base_greek_cfg.mc.compute_rho) {
635 std::cout << "\n\n >>> Computing Rho...\n";
636 const double r = base_greek_cfg.market.r;
637 const double h_r = base_greek_cfg.mc.risk_free_rate_absolute_bump_size;
638
639 KC::UInputs cfg_up = base_greek_cfg;
640 cfg_up.market.r = r + h_r;
641 double p_up = pricing_function(cfg_up, Mem).option_price;
642
643 KC::UInputs cfg_down = base_greek_cfg;
644 cfg_down.market.r = r - h_r;
645 double p_down = pricing_function(cfg_down, Mem).option_price;
646
647 greeks.rho = (p_up - p_down) / (2.0 * h_r);
648 }
649
650 if (base_greek_cfg.mc.compute_theta) {
651 std::cout << "\n\n >>> Computing Theta...\n";
652 const double T = base_greek_cfg.time.t_end; // time to maturity
653 const double h_T = base_greek_cfg.mc.time_absolute_bump_size;
654
655 // Safety check: Ensure the option has more time left than the bump size
656 if (T > h_T) {
657 KC::UInputs cfg_down = base_greek_cfg;
658
659 // Continuous Maturity Shift: Shrink T, but keep N_time_steps the same
660 // NOTE: To price the option from the perspective of tomorrow using a simulation that starts at $0$,
661 // we set the total simulation horizon ($t_{\text{end}}$) to the remaining time: $T - h_T$.
662 // In other words, we are standing at higher unit time (e.g. tomorrow), so, because 1 unit time has elapsed,
663 // the time remaining until maturity is now T−ht
664 cfg_down.time.t_end = T - h_T;
665
666 // Manually compress the continuous grid step to fit the new time, but keep N_time_steps the same
667 cfg_down.time.dt = cfg_down.time.t_end / base_greek_cfg.time.N_time_steps;
668
669 // Now run the pricing function with the compressed grid (the grid has the same shape)
670 double p_down = pricing_function(cfg_down, Mem).option_price;
671
672 // Annualized Theta: (Price with perspective of tomorrow - Price with perspective of today) / time elapsed
673 greeks.theta = (p_down - ref_option_price_data.option_price) / h_T;
674 } else {
675 std::cout <<
676 " [Warning] Option too close to expiry to compute finite-difference Theta. Theta set to zero.\n";
677 greeks.theta = 0.0;
678 }
679 }
680
681 if (base_greek_cfg.mc.compute_vanna) {
682 std::cout << "\n\n >>> Computing Vanna...\n";
683
684 // Get bump on S0
685 const double S0 = base_greek_cfg.market.S0;
686 const double h_S = S0 * base_greek_cfg.mc.spot_price_relative_bump_size;
687 auto h_S_real_scale = h_S;
688 if (base_greek_cfg.mc.normalize_prices) {
689 // Handle price normalization for the denominator
690 h_S_real_scale = base_greek_cfg.get_price_scaling_factor() * base_greek_cfg.mc.
691 spot_price_relative_bump_size;
692 }
693
694 // Get bumped var0
695 const double var0 = base_greek_cfg.market.v0;
696 const double vol0 = std::sqrt(var0);
697 const double h_vol = base_greek_cfg.mc.volatility_absolute_bump_size;
698 const double vol_up = vol0 + h_vol;
699 const double vol_down = vol0 - h_vol;
700 const double var_up = vol_up * vol_up;
701 const double var_down = vol_down * vol_down;
702
703 // 1. Up-Up (Spot Up, Vol Up)
704 KC::UInputs cfg_uu = base_greek_cfg;
705 cfg_uu.market.S0 = S0 + h_S;
706 cfg_uu.market.v0 = var_up;
707 double p_uu = pricing_function(cfg_uu, Mem).option_price;
708
709 // 2. Down-Up (Spot Down, Vol Up)
710 KC::UInputs cfg_du = base_greek_cfg;
711 cfg_du.market.S0 = S0 - h_S;
712 cfg_du.market.v0 = var_up;
713 double p_du = pricing_function(cfg_du, Mem).option_price;
714
715 // 3. Up-Down (Spot Up, Vol Down)
716 KC::UInputs cfg_ud = base_greek_cfg;
717 cfg_ud.market.S0 = S0 + h_S;
718 cfg_ud.market.v0 = var_down;
719 double p_ud = pricing_function(cfg_ud, Mem).option_price;
720
721 // 4. Down-Down (Spot Down, Vol Down)
722 KC::UInputs cfg_dd = base_greek_cfg;
723 cfg_dd.market.S0 = S0 - h_S;
724 cfg_dd.market.v0 = var_down;
725 double p_dd = pricing_function(cfg_dd, Mem).option_price;
726
727 // Vanna 2D Stencil
728 const double inv_denom = 1.0 / (4.0 * h_S_real_scale * h_vol);
729 greeks.vanna = (p_uu - p_du - p_ud + p_dd) * inv_denom;
730 }
731
732 // Eval time and print results
733 Timer.stop_greeks();
734 greeks.print_summary(); // print the summary of the greeks computed
735 }
736
737 // 4. Return combined results
738 Timer.stop_total();
